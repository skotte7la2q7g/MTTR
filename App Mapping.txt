#Requires -Version 5.1
<#
=========================================================================================
 SQL APPLICATION RESILIENCE AND MTTR FRAMEWORK
-----------------------------------------------------------------------------------------
 WHAT THIS PRODUCES
   One self-contained HTML file with four panels, read in this order:

     1. Framework & MTTR     The explanation. What the framework is, the three failure
                             scenarios, the replication mechanisms, how every restore
                             time was arrived at, and the assumptions you own.
     2. Application Overview The application at the top of a relationship map, then one
                             card per environment, then the SQL instances inside it, then
                             the databases on each instance. Click an instance name to
                             jump to its full recovery scope.
     3. SQL Instance MTTR    Pick a SQL instance and see every database on it and how
                             long the whole instance takes to bring back.
     4. Executive Deck       Work in progress. One screen that says what this is.

 SCENARIO RUN-DOWN
   Panels 2 and 3 each carry a scenario drop-down, and the report walks the chosen failure
   step by step: what is rebuilt, which databases are restored in what order, how long
   each step takes and on what evidence, the clock after each step, which applications
   are down, and what the observed backup, recovery-model and replication facts mean.

     Application Overview   Whole application is lost
                            Database corruption - restore from backup (databases you tick)
     SQL Instance MTTR      Whole SQL instance is lost (the VM survives)
                            Database corruption - restore from backup (databases you tick)
                            Whole VM is lost (every SQL instance on it)

 SIMULATOR
   -Simulate builds the whole report from a built-in, made-up estate - no workbook, no
   Excel, no SQL Server, nothing contacted. Every name starts SIM and every page carries a
   SIMULATED banner. Use it to see the report and try every scenario before touching the
   real estate, or to show someone what the output looks like. Its assumption values are
   saved separately, so they never overwrite the ones entered for a real report.

 COUNTS, NOT MARKERS
   Where a set of databases is summarised it is stated as a number and an environment
   spread - "6 databases, PROD 4 / UAT 1 / DEV 1" - never as a row of markers you would
   have to count yourself.

 TWO RULES THE WHOLE REPORT FOLLOWS
   1. Anything not known prints as NULL. Never as zero, never as an average, and never
      dressed up in a phrase. NULL is a fact: it says go and collect this.
   2. Recovery is counted one database at a time, one instance at a time. That is the
      longest it can take, and it is the only figure worth planning against.

 HOW A RESTORE TIME IS ARRIVED AT
   Every database resolves down a three-rung ladder and the rung is printed next to the
   number, everywhere the number appears:

     1. MEASURED   A restore that was actually timed on this database.
     2. DERIVED    This database's own measured backup throughput from msdb, multiplied
                   by a single named restore factor the operator owns.
     3. NULL       No timed restore and no backup history. Nothing is invented.

 INPUTS
   Required : an Excel workbook with sheets 'Prod_DBInventory' and 'NonProd_Inventory'.
              Any other sheet carrying an 'Application Name' column is read as extra
              per-application metadata and shown on the application panel.
   Optional : -CollectSqlEvidence connects read-only to each instance and reads msdb for
              backup and restore history, sys.databases for size and compatibility level,
              and the HA/DR configuration. Without it the report still builds and the
              unmeasured fields read NULL.

 SAFETY
   Every statement issued against SQL Server is a SELECT. Nothing is created, altered or
   dropped. Connections are tagged Application Name=SQL-MTTR-Framework so anyone asking
   "what is this session" can see.

 USAGE
   .\New-MttrFramework.ps1 -Simulate                                       # made-up estate
   .\New-MttrFramework.ps1
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -CollectSqlEvidence
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -CollectSqlEvidence -TestConnectivity
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -Palette Slate    # unbranded copy
=========================================================================================
#>

function Select-ExcelWorkbook {
    Add-Type -AssemblyName System.Windows.Forms
    $dialog = New-Object System.Windows.Forms.OpenFileDialog
    $dialog.Title = 'Select the application / SQL inventory workbook'
    $dialog.Filter = 'Excel workbooks (*.xlsx;*.xlsm;*.xls)|*.xlsx;*.xlsm;*.xls|All files (*.*)|*.*'
    $dialog.FilterIndex = 1
    $dialog.CheckFileExists = $true
    $dialog.CheckPathExists = $true
    $dialog.Multiselect = $false
    $dialog.RestoreDirectory = $true
    if ($dialog.ShowDialog() -ne [System.Windows.Forms.DialogResult]::OK) {
        throw 'No Excel workbook was selected. The report was not generated.'
    }
    return $dialog.FileName
}

function ConvertTo-NormalizedName {
    param([AllowNull()][string]$Value)
    if ([string]::IsNullOrWhiteSpace($Value)) { return '' }
    return (($Value -replace '\s+', ' ').Trim())
}

function Get-HostAndPort {
    param([string]$Instance)
    $value = ConvertTo-NormalizedName $Instance
    $port = 1433
    if ($value -match ',\s*(\d+)\s*$') {
        $port = [int]$Matches[1]
        $value = $value -replace ',\s*\d+\s*$', ''
    }
    $hostName = ($value -split '\\')[0]
    [pscustomobject]@{ Host = $hostName; Port = $port }
}

function Test-TcpEndpoint {
    param([string]$HostName, [int]$Port, [int]$TimeoutMs)
    if ([string]::IsNullOrWhiteSpace($HostName)) { return 'Unknown' }
    $client = [System.Net.Sockets.TcpClient]::new()
    try {
        $task = $client.ConnectAsync($HostName, $Port)
        if (-not $task.Wait($TimeoutMs)) { return 'Disconnected' }
        if ($client.Connected) { return 'Connected' }
        return 'Disconnected'
    }
    catch { return 'Disconnected' }
    finally { $client.Dispose() }
}

function Read-ExcelWorksheet {
    param([Parameter(Mandatory)]$Workbook, [Parameter(Mandatory)][string]$WorksheetName)
    try { $sheet = $Workbook.Worksheets.Item($WorksheetName) }
    catch { throw "Worksheet '$WorksheetName' was not found in the workbook. Expected sheets named 'Prod_DBInventory' and 'NonProd_Inventory'." }

    $used = $sheet.UsedRange
    $values = $used.Value2
    if ($null -eq $values) { return @() }

    $rowCount = $used.Rows.Count
    $columnCount = $used.Columns.Count
    $headers = @{}
    for ($column = 1; $column -le $columnCount; $column++) {
        $name = ConvertTo-NormalizedName ([string]$values[1, $column])
        if ($name) { $headers[$name.ToLowerInvariant()] = $column }
    }

    $rows = [System.Collections.Generic.List[object]]::new()
    for ($rowNumber = 2; $rowNumber -le $rowCount; $rowNumber++) {
        $row = [ordered]@{}
        foreach ($header in $headers.Keys) {
            $row[$header] = ConvertTo-NormalizedName ([string]$values[$rowNumber, $headers[$header]])
        }
        if (($row.Values | Where-Object { $_ }).Count -gt 0) { $rows.Add([pscustomobject]$row) }
    }
    return $rows.ToArray()
}

function Read-ApplicationDetails {
    param([Parameter(Mandatory)]$Workbook, [string]$RequestedWorksheet = '')

    $excluded = @('prod_dbinventory', 'nonprod_inventory')
    $applicationHeaders = @('application name', 'application', 'app name', 'appname')
    $candidateSheets = [System.Collections.Generic.List[object]]::new()

    foreach ($sheet in @($Workbook.Worksheets)) {
        $sheetName = ConvertTo-NormalizedName ([string]$sheet.Name)
        if ($excluded -contains $sheetName.ToLowerInvariant()) { continue }
        if ($RequestedWorksheet -and $sheetName -ne $RequestedWorksheet) { continue }

        $used = $sheet.UsedRange
        if ($used.Rows.Count -lt 2 -or $used.Columns.Count -lt 2) { continue }
        $values = $used.Value2
        $headers = [System.Collections.Generic.List[object]]::new()
        $applicationColumn = 0

        for ($column = 1; $column -le $used.Columns.Count; $column++) {
            $displayName = ConvertTo-NormalizedName ([string]$values[1, $column])
            if (-not $displayName) { continue }
            $headers.Add([pscustomobject]@{ Name = $displayName; Column = $column })
            if ($applicationHeaders -contains $displayName.ToLowerInvariant()) { $applicationColumn = $column }
        }

        if ($applicationColumn -gt 0) {
            $candidateSheets.Add([pscustomobject]@{
                Sheet = $sheet; Name = $sheetName; Values = $values
                Headers = $headers.ToArray(); ApplicationColumn = $applicationColumn
                RowCount = $used.Rows.Count
            })
        }
    }

    if ($RequestedWorksheet -and $candidateSheets.Count -eq 0) {
        throw "Details worksheet '$RequestedWorksheet' was not found, or has no Application Name column."
    }

    $records = [System.Collections.Generic.List[object]]::new()
    foreach ($candidate in $candidateSheets) {
        for ($rowNumber = 2; $rowNumber -le $candidate.RowCount; $rowNumber++) {
            $application = ConvertTo-NormalizedName ([string]$candidate.Values[$rowNumber, $candidate.ApplicationColumn])
            if (-not $application) { continue }
            $fields = [System.Collections.Generic.List[object]]::new()
            foreach ($header in $candidate.Headers) {
                if ($header.Column -eq $candidate.ApplicationColumn) { continue }
                $value = ConvertTo-NormalizedName ([string]$candidate.Values[$rowNumber, $header.Column])
                if (-not $value) { $value = 'NULL' }
                $fields.Add([ordered]@{ label = $header.Name; value = $value; isNull = ($value -eq 'NULL') })
            }
            if ($fields.Count -gt 0) {
                $records.Add([ordered]@{ application = $application; sheet = $candidate.Name; fields = $fields.ToArray() })
            }
        }
    }
    return $records.ToArray()
}

function Get-FieldValue {
    param($Row, [string[]]$Candidates)
    foreach ($candidate in $Candidates) {
        $property = $Row.PSObject.Properties[$candidate.ToLowerInvariant()]
        if ($null -ne $property -and -not [string]::IsNullOrWhiteSpace([string]$property.Value)) {
            return ConvertTo-NormalizedName ([string]$property.Value)
        }
    }
    return ''
}

# =========================================================================================
# REGION  Read-only SQL Server evidence collection
#
#   Everything here degrades. One query failing on one instance costs you that one fact,
#   not the instance, and never the run. What could not be read is reported as
#   "not collected", which is a different statement from "nothing there".
# =========================================================================================

function Invoke-SqlRead {
    param(
        [Parameter(Mandatory)][string]$Instance,
        [string]$Database = 'master',
        [Parameter(Mandatory)][string]$Query,
        [int]$TimeoutSeconds = 20,
        [pscredential]$Credential
    )

    $connectionString = "Server=$Instance;Database=$Database;Application Name=SQL-MTTR-Framework;Connect Timeout=$TimeoutSeconds;"
    if ($Credential) {
        $connectionString += ('User ID={0};Password={1};' -f $Credential.UserName, $Credential.GetNetworkCredential().Password)
    }
    else {
        $connectionString += 'Integrated Security=SSPI;'
    }

    $connection = New-Object System.Data.SqlClient.SqlConnection $connectionString
    try {
        $connection.Open()
        $command = $connection.CreateCommand()
        $command.CommandText = $Query
        $command.CommandTimeout = $TimeoutSeconds
        $table = New-Object System.Data.DataTable
        $reader = $command.ExecuteReader()
        $table.Load($reader)
        $reader.Close()

        $rows = [System.Collections.Generic.List[object]]::new()
        foreach ($dataRow in $table.Rows) {
            $item = [ordered]@{}
            foreach ($column in $table.Columns) {
                $value = $dataRow[$column.ColumnName]
                $item[$column.ColumnName] = $(if ($value -is [System.DBNull]) { $null } else { $value })
            }
            $rows.Add([pscustomobject]$item)
        }
        return $rows.ToArray()
    }
    finally { $connection.Dispose() }
}

# "A network-related or instance-specific error occurred" tells the reader nothing and is the
# single most common thing a first-time runner will see. Name the actual cause and the fix.
function Get-ConnectionDiagnosis {
    param([string]$Message)

    $text = $Message
    $marker = 'argument(s): "'
    $position = $text.IndexOf($marker)
    if ($position -ge 0) { $text = $text.Substring($position + $marker.Length).TrimEnd('"') }
    $text = ($text -replace '\s+', ' ').Trim()

    if ($text -match 'error: 26|Error Locating Server') {
        return 'Named instance could not be resolved. SQL Browser on UDP 1434 is usually blocked - use SERVER,PORT instead of SERVER\INSTANCE.'
    }
    if ($text -match 'error: 40|server was not found') {
        return 'Server not found, or not accepting remote connections. Check the name first, then the firewall on TCP 1433.'
    }
    if ($text -match 'Login failed') {
        return 'Reached the instance but the login was refused. Grant the account VIEW SERVER STATE and read access to msdb.'
    }
    if ($text -match 'timeout expired|pre-login|period of time') {
        return 'No response before the connection timeout expired. Raise -SqlTimeoutSeconds, or the instance is down.'
    }
    if ($text -match 'permission|denied|principal') {
        return 'Reached the instance but the account cannot read the evidence. Needs VIEW SERVER STATE and msdb db_datareader.'
    }
    if ($text.Length -gt 180) { $text = $text.Substring(0, 180) + '...' }
    return $text
}

function Get-SqlInstanceEvidence {
    param(
        [Parameter(Mandatory)][string]$Instance,
        [int]$TimeoutSeconds,
        [pscredential]$Credential,
        [int]$WindowDays,
        [string[]]$CommandLogDatabase
    )

    $evidence = [ordered]@{
        instance = $Instance; status = 'Unreachable'; message = ''
        machine = ''; version = ''; productLevel = ''; edition = ''
        isClustered = $false; isHadrEnabled = $false
        modules = [ordered]@{}
        databases = @{}
    }

    # --- Instance identity. If this fails the instance is genuinely unreachable. -------------
    try {
        $facts = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT
    CAST(SERVERPROPERTY('MachineName')    AS nvarchar(200)) AS MachineName,
    CAST(SERVERPROPERTY('ProductVersion') AS nvarchar(50))  AS ProductVersion,
    CAST(SERVERPROPERTY('ProductLevel')   AS nvarchar(50))  AS ProductLevel,
    CAST(SERVERPROPERTY('Edition')        AS nvarchar(200)) AS Edition,
    CAST(ISNULL(SERVERPROPERTY('IsClustered'),0)   AS int)  AS IsClustered,
    CAST(ISNULL(SERVERPROPERTY('IsHadrEnabled'),0) AS int)  AS IsHadrEnabled;
"@
        if ($facts.Count -gt 0) {
            $evidence.status = 'Collected'
            $evidence.machine = [string]$facts[0].MachineName
            $evidence.version = [string]$facts[0].ProductVersion
            $evidence.productLevel = [string]$facts[0].ProductLevel
            $evidence.edition = [string]$facts[0].Edition
            $evidence.isClustered = ([int]$facts[0].IsClustered -eq 1)
            $evidence.isHadrEnabled = ([int]$facts[0].IsHadrEnabled -eq 1)
        }
    }
    catch {
        $evidence.message = Get-ConnectionDiagnosis $_.Exception.Message
        return $evidence
    }

    function Add-DbFact {
        param($Store, [string]$Name, [hashtable]$Values)
        if ([string]::IsNullOrWhiteSpace($Name)) { return }
        $key = $Name.ToUpperInvariant()
        if (-not $Store.ContainsKey($key)) { $Store[$key] = [ordered]@{ name = $Name } }
        foreach ($entry in $Values.GetEnumerator()) { $Store[$key][$entry.Key] = $entry.Value }
    }

    # --- Size, compatibility level, recovery model, state -------------------------------------
    try {
        $databases = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  d.name                                                       AS DatabaseName,
        d.compatibility_level                                        AS CompatibilityLevel,
        d.recovery_model_desc                                        AS RecoveryModel,
        d.state_desc                                                 AS State,
        d.is_published, d.is_subscribed, d.is_merge_published,
        CAST(SUM(CAST(mf.size AS bigint)) * 8.0 / 1048576.0 AS decimal(18,2)) AS SizeGB
FROM    sys.databases d
LEFT JOIN sys.master_files mf ON mf.database_id = d.database_id
GROUP BY d.name, d.compatibility_level, d.recovery_model_desc, d.state_desc,
         d.is_published, d.is_subscribed, d.is_merge_published;
"@
        foreach ($database in $databases) {
            Add-DbFact $evidence.databases ([string]$database.DatabaseName) @{
                sizeGb        = $(if ($null -ne $database.SizeGB) { [double]$database.SizeGB } else { $null })
                compatibility = $(if ($null -ne $database.CompatibilityLevel) { [int]$database.CompatibilityLevel } else { $null })
                recoveryModel = [string]$database.RecoveryModel
                state         = [string]$database.State
                isReplicated  = (([int]$database.is_published -eq 1) -or ([int]$database.is_subscribed -eq 1) -or ([int]$database.is_merge_published -eq 1))
            }
        }
        $evidence.modules['Databases'] = 'Collected'
    }
    catch { $evidence.modules['Databases'] = "Failed: $($_.Exception.Message)" }

    # --- Measured backup throughput. This is the honest basis for a restore estimate. ---------
    #     Throughput is measured on this database, on this instance, on this storage.
    #     The last full and last log backup dates count every backup; only the throughput
    #     average skips the sub-second ones, which have no measurable duration.
    try {
        $backups = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  bs.database_name                                                       AS DatabaseName,
        MAX(CASE WHEN bs.type = 'D' THEN bs.backup_finish_date END)            AS LastFullBackup,
        MAX(CASE WHEN bs.type = 'L' THEN bs.backup_finish_date END)            AS LastLogBackup,
        SUM(CASE WHEN bs.type = 'D' AND bs.backup_size > 0
                  AND DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) > 0
                 THEN 1 ELSE 0 END)                                            AS Samples,
        AVG(CASE WHEN bs.type = 'D' AND bs.backup_size > 0
                  AND DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) > 0
                 THEN CAST(DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) AS bigint) END) AS AvgSeconds,
        AVG(CASE WHEN bs.type = 'D' AND bs.backup_size > 0
                  AND DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) > 0
                 THEN CAST(bs.backup_size / 1048576.0 AS float) END)           AS AvgMegabytes
FROM    msdb.dbo.backupset bs
WHERE   bs.type IN ('D', 'L')
  AND   bs.backup_finish_date > DATEADD(DAY, -$WindowDays, GETDATE())
GROUP BY bs.database_name;
"@
        foreach ($backup in $backups) {
            $seconds = $(if ($null -ne $backup.AvgSeconds) { [double]$backup.AvgSeconds } else { 0.0 })
            $megabytes = $(if ($null -ne $backup.AvgMegabytes) { [double]$backup.AvgMegabytes } else { 0.0 })
            $throughput = $null
            if ($seconds -gt 0 -and $megabytes -gt 0) { $throughput = [math]::Round($megabytes / $seconds, 2) }
            Add-DbFact $evidence.databases ([string]$backup.DatabaseName) @{
                lastFullBackup      = $(if ($null -ne $backup.LastFullBackup) { ([datetime]$backup.LastFullBackup).ToString('yyyy-MM-dd') } else { $null })
                lastLogBackup       = $(if ($null -ne $backup.LastLogBackup) { ([datetime]$backup.LastLogBackup).ToString('yyyy-MM-dd HH:mm') } else { $null })
                backupSamples       = [int]$backup.Samples
                backupThroughputMBs = $throughput
                backupMinutes       = $(if ($seconds -gt 0) { [math]::Round($seconds / 60.0, 1) } else { $null })
            }
        }
        $evidence.modules['Backup history'] = 'Collected'
    }
    catch { $evidence.modules['Backup history'] = "Failed: $($_.Exception.Message)" }

    # --- Has this database ever actually been restored, and when ------------------------------
    try {
        $restores = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  rh.destination_database_name AS DatabaseName,
        MAX(rh.restore_date)         AS LastRestore,
        COUNT(*)                     AS RestoreCount
FROM    msdb.dbo.restorehistory rh
WHERE   rh.restore_date > DATEADD(DAY, -$WindowDays, GETDATE())
GROUP BY rh.destination_database_name;
"@
        foreach ($restore in $restores) {
            Add-DbFact $evidence.databases ([string]$restore.DatabaseName) @{
                lastRestore  = $(if ($null -ne $restore.LastRestore) { ([datetime]$restore.LastRestore).ToString('yyyy-MM-dd') } else { $null })
                restoreCount = [int]$restore.RestoreCount
            }
        }
        $evidence.modules['Restore history'] = 'Collected'
    }
    catch { $evidence.modules['Restore history'] = "Failed: $($_.Exception.Message)" }

    # --- Last clean DBCC CHECKDB. Says which backups were taken from a database known to be ---
    #     clean. SQL Server 2016 SP2 and later report it; older builds return NULL, which is
    #     left as NULL. 1900-01-01 is SQL Server's way of saying no clean check on record.
    try {
        $checks = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  d.name                                                                  AS DatabaseName,
        CAST(DATABASEPROPERTYEX(d.name, 'LastGoodCheckDbTime') AS datetime)     AS LastGoodCheckDb
FROM    sys.databases d
WHERE   d.state = 0;
"@
        foreach ($check in $checks) {
            if ($null -eq $check.LastGoodCheckDb) { continue }
            $when = [datetime]$check.LastGoodCheckDb
            Add-DbFact $evidence.databases ([string]$check.DatabaseName) @{
                lastGoodCheckDb = $(if ($when.Year -le 1900) { 'Never' } else { $when.ToString('yyyy-MM-dd') })
            }
        }
        $evidence.modules['Integrity checks'] = 'Collected'
    }
    catch { $evidence.modules['Integrity checks'] = "Failed: $($_.Exception.Message)" }

    # --- A genuinely timed restore, if anything on this estate logs one ------------------------
    #     Ola Hallengren's maintenance solution writes CommandLog with StartTime/EndTime.
    #     Where it exists this is the only truly measured restore duration available.
    $commandLogFound = $false
    foreach ($logDatabase in $CommandLogDatabase) {
        if ($commandLogFound) { break }
        try {
            $timed = Invoke-SqlRead -Instance $Instance -Database $logDatabase -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
IF OBJECT_ID('dbo.CommandLog') IS NOT NULL
    SELECT  cl.DatabaseName                                                   AS DatabaseName,
            COUNT(*)                                                          AS Samples,
            AVG(CAST(DATEDIFF(SECOND, cl.StartTime, cl.EndTime) AS bigint))   AS AvgSeconds,
            MAX(cl.EndTime)                                                   AS LastTimedRestore
    FROM    dbo.CommandLog cl
    WHERE   cl.CommandType LIKE 'RESTORE%'
      AND   cl.EndTime IS NOT NULL
      AND   cl.StartTime > DATEADD(DAY, -$WindowDays, GETDATE())
      AND   DATEDIFF(SECOND, cl.StartTime, cl.EndTime) > 0
    GROUP BY cl.DatabaseName;
"@
            if ($timed.Count -gt 0) {
                $commandLogFound = $true
                foreach ($entry in $timed) {
                    Add-DbFact $evidence.databases ([string]$entry.DatabaseName) @{
                        timedRestoreMinutes = [math]::Round(([double]$entry.AvgSeconds) / 60.0, 1)
                        timedRestoreSamples = [int]$entry.Samples
                        timedRestoreSource  = "CommandLog in [$logDatabase]"
                    }
                }
                $evidence.modules['Timed restores'] = "Collected from [$logDatabase]"
            }
        }
        catch { }
    }
    if (-not $commandLogFound -and -not $evidence.modules.Contains('Timed restores')) {
        $evidence.modules['Timed restores'] = 'No timed-restore log found'
    }

    # --- Recovery mechanism. This decides whether a restore is even the recovery path. ---------
    $mechanisms = @{}
    if ($evidence.isClustered) { $mechanisms['*'] = 'Failover Cluster Instance' }

    if ($evidence.isHadrEnabled) {
        try {
            $availabilityGroups = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  DB_NAME(drs.database_id)   AS DatabaseName,
        ag.name                    AS AgName,
        ar.availability_mode_desc  AS AvailabilityMode
FROM    sys.dm_hadr_database_replica_states drs
JOIN    sys.availability_replicas ar ON ar.replica_id = drs.replica_id
JOIN    sys.availability_groups   ag ON ag.group_id  = ar.group_id
WHERE   drs.is_local = 1;
"@
            foreach ($group in $availabilityGroups) {
                $mode = [string]$group.AvailabilityMode
                $label = "Always On AG '$($group.AgName)'"
                if ($mode -match 'SYNCHRONOUS') { $label += ' (synchronous)' } elseif ($mode) { $label += ' (asynchronous)' }
                $mechanisms[([string]$group.DatabaseName).ToUpperInvariant()] = $label
            }
            $evidence.modules['Availability groups'] = 'Collected'
        }
        catch { $evidence.modules['Availability groups'] = "Failed: $($_.Exception.Message)" }
    }

    try {
        $logShipped = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT primary_database AS DatabaseName FROM msdb.dbo.log_shipping_primary_databases;
"@
        foreach ($shipped in $logShipped) {
            $key = ([string]$shipped.DatabaseName).ToUpperInvariant()
            if (-not $mechanisms.ContainsKey($key)) { $mechanisms[$key] = 'Log shipping (primary)' }
        }
    }
    catch { }

    try {
        $mirrored = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT DB_NAME(database_id) AS DatabaseName, mirroring_state_desc AS MirrorState
FROM   sys.database_mirroring WHERE mirroring_guid IS NOT NULL;
"@
        foreach ($mirror in $mirrored) {
            $key = ([string]$mirror.DatabaseName).ToUpperInvariant()
            if (-not $mechanisms.ContainsKey($key)) { $mechanisms[$key] = "Database mirroring ($($mirror.MirrorState))" }
        }
    }
    catch { }

    foreach ($key in @($evidence.databases.Keys)) {
        $mechanism = ''
        if ($mechanisms.ContainsKey($key)) { $mechanism = $mechanisms[$key] }
        elseif ($mechanisms.ContainsKey('*')) { $mechanism = $mechanisms['*'] }
        elseif ($evidence.databases[$key].Contains('isReplicated') -and $evidence.databases[$key]['isReplicated']) { $mechanism = 'Transactional / merge replication' }
        else { $mechanism = 'Backup and restore only' }
        $evidence.databases[$key]['recoveryMechanism'] = $mechanism
    }

    return $evidence
}

# REGION  Simulator
#
#   A made-up estate, built in exactly the shapes the workbook reader and the SQL collector
#   produce, so everything after this point runs as it does for real. It is built to show
#   every case the scenario run-down can meet: a VM carrying two SQL instances, a
#   synchronous availability group, a failover cluster, log shipping, measured, derived and
#   NULL restore times, SIMPLE recovery, a database with no backup, one never integrity
#   checked, one SUSPECT, a database shared by three applications, an unreachable instance
#   and one nobody has collected from. Dates are relative to today, so it never goes stale.
# =========================================================================================

function Get-SimulatedEstate {
    $now = Get-Date
    function Get-SimDate {
        param($Days, [string]$Format = 'yyyy-MM-dd')
        if ($null -eq $Days) { return $null }
        $now.AddDays(-$Days).ToString($Format)
    }

    # Collected instances get a full evidence record; SIMUAT01 fails to connect; SIMDEV01 was
    # never collected, so everything about it comes from the "workbook" alone.
    $simInstances = @(
        @{ Name = 'SIMPRD01\CLAIMS'; Server = 'SIMPRD01'; Version = '13.0.6300.2'; Level = 'SP3'; Clustered = $false; Hadr = $false; Reach = 'Connected';    Status = 'Collected' }
        @{ Name = 'SIMPRD01\POLICY'; Server = 'SIMPRD01'; Version = '13.0.6300.2'; Level = 'SP3'; Clustered = $false; Hadr = $false; Reach = 'Connected';    Status = 'Collected' }
        @{ Name = 'SIMPRD02';        Server = 'SIMPRD02'; Version = '15.0.4355.3'; Level = 'RTM'; Clustered = $false; Hadr = $true;  Reach = 'Connected';    Status = 'Collected' }
        @{ Name = 'SIMPRD03';        Server = 'SIMPRD03'; Version = '16.0.4135.4'; Level = 'RTM'; Clustered = $true;  Hadr = $false; Reach = 'Connected';    Status = 'Collected' }
        @{ Name = 'SIMUAT01';        Server = 'SIMUAT01'; Version = '2016';        Level = '';    Clustered = $false; Hadr = $false; Reach = 'Disconnected'; Status = 'Unreachable' }
        @{ Name = 'SIMDEV01';        Server = 'SIMDEV01'; Version = '2019';        Level = '';    Clustered = $false; Hadr = $false; Reach = 'Not tested';   Status = 'Not collected' }

    function New-SimDb {

           State = $State; FullDays = $FullDays; LogHours = $LogHours; Mbps = $Mbps; CheckDays = $CheckDays
           TimedMinutes = $TimedMinutes; Mechanism = $Mechanism; SheetRestore = $SheetRestore }
    }
    $agSync = "Always On AG 'SIM_AG_PAY' (synchronous)"
    $simDatabases = @(
        (New-SimDb -Apps 'Claims Portal' -Instance 'SIMPRD01\CLAIMS' -Name 'ClaimsDB' -Env 'PROD' -Gb 120 -Compat 130 -Mbps 140 -CheckDays 3 -TimedMinutes 95)
        (New-SimDb -Apps 'Claims Portal' -Instance 'SIMPRD01\CLAIMS' -Name 'ClaimsArchive' -Env 'PROD' -Gb 340 -Compat 110 -Model 'SIMPLE' -FullDays 2 -LogHours $null -Mbps 180 -CheckDays 'Never')
        (New-SimDb -Apps 'Claims Portal' -Instance 'SIMPRD01\CLAIMS' -Name 'ClaimsDocs' -Env 'PROD' -Gb 55 -Compat 130 -FullDays $null -LogHours $null -CheckDays 20)
        (New-SimDb -Apps 'Claims Portal', 'Policy Admin', 'Payments Hub' -Instance 'SIMPRD01\CLAIMS' -Name 'RefData' -Env 'PROD' -Gb 8 -Compat 130 -Mbps 90 -CheckDays 6)
        (New-SimDb -Apps 'Policy Admin' -Instance 'SIMPRD01\POLICY' -Name 'PolicyCore' -Env 'PROD' -Gb 210 -Compat 130 -Mbps 110 -CheckDays 4 -TimedMinutes 240 -Mechanism 'Log shipping (primary)')
        (New-SimDb -Apps 'Policy Admin' -Instance 'SIMPRD01\POLICY' -Name 'PolicyRating' -Env 'PROD' -Gb 40 -Compat 120 -Mbps 150 -CheckDays 4)
        (New-SimDb -Apps 'Customer 360' -Instance 'SIMPRD01\POLICY' -Name 'CustMaster' -Env 'PROD' -Gb 95 -Compat 100 -Model 'SIMPLE' -FullDays 4 -LogHours $null -Mbps 120 -CheckDays 30)
        (New-SimDb -Apps 'Payments Hub' -Instance 'SIMPRD02' -Name 'PaymentsDB' -Env 'PROD' -Gb 60 -Compat 150 -FullDays 0.5 -LogHours 0.25 -Mbps 200 -CheckDays 2 -Mechanism $agSync)
        (New-SimDb -Apps 'Payments Hub' -Instance 'SIMPRD02' -Name 'PayLedger' -Env 'PROD' -Gb 150 -Compat 150 -FullDays 0.5 -LogHours 0.25 -Mbps 160 -CheckDays 2 -TimedMinutes 110 -Mechanism $agSync)
        (New-SimDb -Apps 'HR Payroll' -Instance 'SIMPRD02' -Name 'PayrollDB' -Env 'PROD' -Gb 25 -Compat 140 -State 'SUSPECT' -LogHours 3 -Mbps 100 -CheckDays $null)
        (New-SimDb -Apps 'Customer 360' -Instance 'SIMPRD03' -Name 'CustEvents' -Env 'PROD' -Gb 480 -Compat 160 -Mbps 250 -CheckDays 7)
        (New-SimDb -Apps 'HR Payroll' -Instance 'SIMPRD03' -Name 'HRCore' -Env 'PROD' -Gb 30 -Compat 160 -Mbps 120 -CheckDays 7)
        (New-SimDb -Apps 'Claims Portal' -Instance 'SIMUAT01' -Name 'ClaimsDB_UAT' -Env 'UAT' -Gb 60 -SheetRestore '45')
        (New-SimDb -Apps 'Policy Admin' -Instance 'SIMUAT01' -Name 'PolicyCore_UAT' -Env 'UAT' -Gb 90 -SheetRestore '70')
        (New-SimDb -Apps 'Claims Portal' -Instance 'SIMDEV01' -Name 'ClaimsDB_DEV' -Env 'DEV' -Gb 12 -SheetRestore '10')

    $nonProdRows = [System.Collections.Generic.List[object]]::new()
    $evidence = @{}
    $connectivity = @{}

    foreach ($i in $simInstances) {
        $connectivity[$i.Name] = $i.Reach
        if ($i.Status -eq 'Not collected') { continue }
        $evidence[$i.Name] = [ordered]@{
            instance = $i.Name; status = $i.Status; message = ''
            machine = $i.Server; version = ''; productLevel = ''; edition = ''
            isClustered = $i.Clustered; isHadrEnabled = $i.Hadr
            modules = [ordered]@{}; databases = @{}

        if ($i.Status -eq 'Unreachable') {
            $evidence[$i.Name].message = 'Server not found, or not accepting remote connections. Check the name first, then the firewall on TCP 1433.'
        }
        else {
            $evidence[$i.Name].version = $i.Version
            $evidence[$i.Name].productLevel = $i.Level
            $evidence[$i.Name].edition = 'Enterprise Edition (64-bit)'
            $evidence[$i.Name].modules['Simulation'] = 'Simulated'
        }
    }

    foreach ($db in $simDatabases) {
        $instance = $simInstances | Where-Object { $_.Name -eq $db.Instance }
        foreach ($app in $db.Apps) {
            $row = [pscustomobject][ordered]@{
                'application name'                = $app
                'instance name'                   = $db.Instance
                'servername'                      = $instance.Server
                'databasename'                    = $db.Name
                'environment'                     = $db.Env
                'database size (gb)'              = [string]$db.Gb
                'ha/dr method'                    = ''
                'last restore duration (minutes)' = $db.SheetRestore
                'sql version'                     = $instance.Version
            }
            if ($db.Env -eq 'PROD') { $prodRows.Add($row) } else { $nonProdRows.Add($row) }
        }

        if ($instance.Status -ne 'Collected') { continue }
        $mechanism = $(if ($db.Mechanism) { $db.Mechanism } elseif ($instance.Clustered) { 'Failover Cluster Instance' } else { 'Backup and restore only' })
        $fact = [ordered]@{
            name = $db.Name; sizeGb = $db.Gb; compatibility = $db.Compat; recoveryModel = $db.Model; state = $db.State
            isReplicated = $false; recoveryMechanism = $mechanism
            lastFullBackup = Get-SimDate $db.FullDays
            lastLogBackup = $(if ($null -ne $db.LogHours) { Get-SimDate ($db.LogHours / 24) 'yyyy-MM-dd HH:mm' } else { $null })
            backupThroughputMBs = $(if ($null -ne $db.FullDays -and $null -ne $db.Mbps) { [double]$db.Mbps } else { $null })
            backupMinutes = $(if ($null -ne $db.FullDays -and $null -ne $db.Mbps) { [math]::Round($db.Gb * 1024 / $db.Mbps / 60, 1) } else { $null })
            lastGoodCheckDb = $(if ('Never' -eq $db.CheckDays) { 'Never' } else { Get-SimDate $db.CheckDays })
            lastRestore = $(if ($null -ne $db.TimedMinutes) { Get-SimDate 30 } else { $null })
        }
        if ($null -ne $db.TimedMinutes) {
            $fact.timedRestoreMinutes = [double]$db.TimedMinutes
            $fact.timedRestoreSamples = 2
            $fact.timedRestoreSource = 'CommandLog in [DBA] (simulated)'
        }
        $evidence[$db.Instance].databases[$db.Name.ToUpperInvariant()] = $fact
    }

    $details = @(
        [ordered]@{ application = 'Claims Portal'; sheet = 'App_Details (simulated)'; fields = @(
            [ordered]@{ label = 'Business owner'; value = 'Claims Operations'; isNull = $false }
            [ordered]@{ label = 'Criticality'; value = 'Tier 1'; isNull = $false }
            [ordered]@{ label = 'RTO agreed'; value = 'NULL'; isNull = $true }) }
        [ordered]@{ application = 'Payments Hub'; sheet = 'App_Details (simulated)'; fields = @(
            [ordered]@{ label = 'Business owner'; value = 'Treasury'; isNull = $false }
            [ordered]@{ label = 'Criticality'; value = 'Tier 1'; isNull = $false }
            [ordered]@{ label = 'RTO agreed'; value = '4 hours'; isNull = $false }) }
    )

    [pscustomobject]@{
        ProdRows = $prodRows.ToArray(); NonProdRows = $nonProdRows.ToArray()
        Details = $details; Evidence = $evidence; Connectivity = $connectivity
    }
}

$inventory = [System.Collections.Generic.List[object]]::new()
function Add-InventoryRows {
    param(
        [Parameter(Mandatory)][object[]]$Rows,
        [Parameter(Mandatory)][string]$DefaultEnvironment
    )
    foreach ($row in $Rows) {
        $application = Get-FieldValue $row @('application name', 'application', 'app name', 'appname')
        $instance = Get-FieldValue $row @('instance name', 'servername')
        $server = Get-FieldValue $row @('servername')
        $database = Get-FieldValue $row @('databasename', 'database name')
        $environment = Get-FieldValue $row @('environment')
        if (-not $environment) { $environment = $DefaultEnvironment }
        if (-not $server) { $server = (Get-HostAndPort $instance).Host }

        # Hand-captured values from the workbook. These fill gaps the SQL collection leaves,
        # and are overridden by anything actually measured on the instance.
        $sheetSize = Get-FieldValue $row @('database size', 'database size (gb)', 'size gb', 'size (gb)', 'size')
        $sheetCompat = Get-FieldValue $row @('compatibility level', 'compatibility')
        $sheetMechanism = Get-FieldValue $row @('ha/dr method', 'hadr method', 'dr method', 'data replication method', 'replication method')
        $sheetRestore = Get-FieldValue $row @('last restore duration (minutes)', 'restore duration (minutes)', 'measured restore minutes')
        $sheetVersion = Get-FieldValue $row @('sql version', 'sql server version', 'version')

        $inventory.Add([pscustomobject]@{
            Application     = $(if ($application) { $application } else { 'Unattributed' })
            Environment     = $(if ($environment) { $environment.ToUpperInvariant() } else { 'UNKNOWN' })
            Server          = $(if ($server) { $server } else { '' })
            Instance        = $(if ($instance) { $instance } else { '' })
            Database        = $(if ($database) { $database } else { '' })
            SheetSize       = $sheetSize
            SheetCompat     = $sheetCompat
            SheetMechanism  = $sheetMechanism
            SheetRestore    = $sheetRestore
            SheetVersion    = $sheetVersion
        })
    }
}

function Get-DatabaseEvidence {
    param([string]$Instance, [string]$Database)
    if (-not $evidenceByInstance.ContainsKey($Instance)) { return $null }
    $store = $evidenceByInstance[$Instance].databases
    $key = $Database.ToUpperInvariant()
    if ($store.ContainsKey($key)) { return $store[$key] }
    return $null
}

function Get-EvidenceValue {
    param($Store, [string]$Key)
    if ($null -eq $Store) { return $null }
    if ($Store.Contains($Key)) { return $Store[$Key] }
    return $null
}

    i.siblings = 0;
  });
  /* Two named instances on one host are still two instances, but they share a machine.
     The list says so, so nobody reads them as unrelated. */
  const byServer = new Map();
  map.forEach(i => { const s = String(i.server || '').toUpperCase(); byServer.set(s, (byServer.get(s) || 0) + 1); });
  map.forEach(i => { i.siblings = (byServer.get(String(i.server || '').toUpperCase()) || 1) - 1; });
  return map;
})();

const applications = (() => {
  const map = new Map();
  raw.forEach(r => {
    if (!map.has(r.application)) map.set(r.application, { name:r.application, rows:[], dbKeys:new Set() });
    const a = map.get(r.application);
    a.rows.push(r);
    a.dbKeys.add(dbKey(r));
  });
  map.forEach(a => { a.dbs = [...a.dbKeys].map(k => databases.get(k)); });
  return map;
})();

/* ---------------------------------------------------------------------------
   The evidence ladder: measured, derived, or NULL. Nothing is invented.
--------------------------------------------------------------------------- */
function restoreOf(db) {
  if (db.timedRestoreMinutes != null) {
    const n = db.timedRestoreSamples || 1;
    return { minutes: db.timedRestoreMinutes, basis: 'measured', label: 'Measured',
             detail: `Timed restore, ${n} run${n === 1 ? '' : 's'}${db.timedRestoreSource ? ' - ' + db.timedRestoreSource : ''}` };
  }
  const factor = num(cfg.factor);
  if (db.backupThroughputMBs != null && db.sizeGb != null && factor !== null) {
    return { minutes: (db.sizeGb * 1024 / db.backupThroughputMBs / 60) * factor, basis: 'derived', label: 'Derived',
             detail: `${db.sizeGb.toFixed(1)} GB at a measured ${db.backupThroughputMBs} MB/s backup throughput, x ${factor} restore factor` };
  }
  return { minutes: null, basis: 'none', label: 'NULL',
           detail: 'No timed restore and no backup history for this database. Run collection to measure it.' };
}

const basisBadge = r => `<span class="badge b-${r.basis}" title="${esc(r.detail)}">${esc(r.label)}</span>`;

/* One database at a time. That is the longest it can take, and it is the figure to plan against. */
function restoreWorkload(dbs) {
  const parts = dbs.map(restoreOf);
  const known = parts.filter(p => p.minutes !== null).map(p => p.minutes);
  return {
    total: known.length ? sum(known) : null,
    counted: known.length,
    unquantified: parts.length - known.length,
    mix: { measured: parts.filter(p => p.basis === 'measured').length,
           derived:  parts.filter(p => p.basis === 'derived').length,
           none:     parts.filter(p => p.basis === 'none').length }
  };
}

/* One leg is one instance's share of a recovery: a rebuild where one is needed, the restores,
   then validation. A leg needing a rebuild nobody has sized, or with no restore figure at all,
   is NULL. The headline figures and the scenario run-downs both use this, so they agree. */
function buildMinutesFor(environment) {
  const build = buildHours(environment);
  return build === null ? null : build * 60;
}
function vmMinutesFor(environment) {
  const vm = vmHours(environment);
  return vm === null ? null : vm * 60;
}
function legTotal(work, buildMinutes, needsBuild) {
  if (work.total === null) return null;
  if (needsBuild && buildMinutes === null) return null;
  return work.total + (needsBuild ? buildMinutes : 0) + (num(cfg.validation) || 0) * 60;
}

/* Scenario 3 - the whole instance is gone. */
function instanceMttr(inst) {
  const work = restoreWorkload(inst.databases);
  const buildMinutes = buildMinutesFor(inst.environment);
  return { ...work, buildMinutes, complete: legTotal(work, buildMinutes, true) };
}

/* Scenario 2 - the application is gone. Only its own databases are in scope, even where
   the instance is shared, but the rebuild of each instance it sits on still applies. */
function applicationMttr(app, environmentFilter) {
  const dbs = app.dbs.filter(d => environmentFilter === 'ALL' || d.environments.has(environmentFilter));
  const byInstance = new Map();
  dbs.forEach(d => {
    if (!byInstance.has(d.instance)) byInstance.set(d.instance, []);
    byInstance.get(d.instance).push(d);
  });

  const legs = [...byInstance.entries()].sort((a, b) => byName(a[0], b[0])).map(([instanceName, instanceDbs]) => {
    const inst = instances.get(instanceName);
    const work = restoreWorkload(instanceDbs);
    const buildMinutes = buildMinutesFor(inst.environment);
    return { instance: inst, dbs: instanceDbs.slice().sort((a, b) => byName(a.database, b.database)),
             work, buildMinutes, total: legTotal(work, buildMinutes, true) };
  });

  const totals = legs.map(l => l.total).filter(t => t !== null);
  const complete = totals.length === legs.length && legs.length > 0 ? sum(totals) : null;

  const mix = { measured: 0, derived: 0, none: 0 };
  legs.forEach(l => Object.keys(mix).forEach(k => mix[k] += l.work.mix[k]));

  return { dbs, legs, complete, mix,
           collateral: uniq(legs.flatMap(l => l.instance.appList)).filter(a => a !== app.name).sort(byName) };
}

--------------------------------------------------------------------------- */

/* What kind of second copy a database has. Collected text and workbook text both land here. */
function mechClass(mechanism) {
  const t = String(mechanism || '').trim().toLowerCase();
  if (!t) return 'unknown';
  if (/mirror/.test(t)) return 'mirror';
  if (/log ?shipping/.test(t)) return 'logship';
  if (/always ?on|availability group|\bag\b/.test(t)) {
    if (/asynchronous|\basync/.test(t)) return 'async';
    return /synchronous|\bsync/.test(t) ? 'sync' : 'async';
  }
  if (/cluster|\bfci\b/.test(t)) return 'fci';
  if (/replication/.test(t)) return 'repl';
  if (/backup|standalone|^none$|^n\/?a$/.test(t)) return 'backup';
  return 'unknown';
}
const HAS_REPLICA = ['sync', 'async', 'mirror', 'logship'];

/* What a second copy does and does not do for each kind of failure. */
const MECH_NOTE = {

    async:   ['ok', 'An asynchronous replica elsewhere can take this off the critical path, with some data loss.'],
    mirror:  ['ok', 'A mirror elsewhere can take this database off the critical path.'],
    logship: ['ok', 'A log shipping secondary can take this off the critical path, behind by the logs not yet restored.'],
    fci:     ['info', 'The cluster covers a failed node, not lost storage. If the storage has gone, the restore applies.'],
    repl:    ['info', 'Replication is not recovery. Restore, then re-initialise replication.'],
    backup:  ['info', 'No replica. Restore is the plan.'],
    unknown: ['none', 'Replication / HA is NULL. Treat restore as the plan until it is confirmed.']
  },
  corrupt: {

    async:   ['warn', 'The asynchronous replica may hold the same corruption. Check it with DBCC CHECKDB before failing over; otherwise restore.'],
    mirror:  ['warn', 'The mirror may hold the same corruption. Check it before failing over; otherwise restore.'],
    logship: ['warn', 'The log shipping secondary may still be clean if it has not applied the damaged log. Check it before a role change.'],
    fci:     ['info', 'The cluster runs on one copy of the storage, so failover does not help. Restore is the path.'],
    repl:    ['info', 'Replication is not recovery. Restore, then re-initialise replication.'],

const plural = (n, one, many) => `${n} ${n === 1 ? one : (many || one + 's')}`;
const dbName = d => `<b>${esc(d.database)}</b> ${envTag(d.environment)}`;
const nameList = (items, max) => items.slice(0, max).join(', ') + (items.length > max ? ` and ${items.length - max} more` : '');
const clockText = m => m <= 0 ? '0 h' : hoursText(m);

/* Evidence was only looked for where collection reached the instance. "None found" and
   "never looked" are different statements, and the run-down keeps them apart. */
const wasCollected = d => { const i = instances.get(d.instance); return !!(i && i.meta && i.meta.status === 'Collected'); };
const NOT_COLLECTED = '<span class="nul">Not collected</span>';
const seen = (d, value, show) => (value !== null && value !== undefined && value !== '')
  ? show(value) : (wasCollected(d) ? NUL : NOT_COLLECTED);
const backupCell = (d, value, severe) => value ? `<span class="mono">${esc(value)}</span>`
  : (wasCollected(d) ? `<span class="${severe ? 'v-bad' : ''}">None in ${WINDOW_DAYS} days</span>` : NOT_COLLECTED);

/* Production first, then by instance name, then by database name. */
function groupByInstance(dbs) {
  const m = new Map();
  dbs.forEach(d => { if (!m.has(d.instance)) m.set(d.instance, []); m.get(d.instance).push(d); });
  return [...m.entries()]
    .map(([name, list]) => [instances.get(name), list.slice().sort((a, b) => byName(a.database, b.database))])
    .sort((a, b) => envRank(a[0].environment) - envRank(b[0].environment) || byName(a[0].instance, b[0].instance));
}
const defaultPick = pool => { const first = groupByInstance(pool).flatMap(([, list]) => list)[0]; return first ? [first.key] : []; };

function runScenario(kind, dbs) {
  const spec = SCENARIOS[kind];
  const validation = num(cfg.validation);
  const legs = groupByInstance(dbs).map(([inst, list]) => ({ inst, dbs: list }));

  const steps = [];
  let clock = 0, open = false;
  const add = step => {
    step.start = clock;
    if (step.minutes === null) open = true; else clock += step.minutes;
    step.end = clock;
    step.open = open;
    steps.push(step);
  };

  /* A lost VM is rebuilt once, before anything else can start. Its environment is the most
     critical one among the instances it carries. */
  let vm = null;
  if (spec.vm && legs.length) {
    const environment = legs[0].inst.environment;
    vm = { name: legs[0].inst.server, environment, minutes: vmMinutesFor(environment) };
    add({ kind: 'vm', leg: null, vm, minutes: vm.minutes });
  }

  legs.forEach(leg => {
    leg.buildMinutes = buildMinutesFor(leg.inst.environment);
    leg.work = restoreWorkload(leg.dbs);
    leg.total = legTotal(leg.work, leg.buildMinutes, spec.rebuild);
    if (spec.rebuild) add({ kind: 'rebuild', leg, minutes: leg.buildMinutes });
    leg.dbs.forEach(d => { const r = restoreOf(d); add({ kind: 'restore', leg, db: d, r, minutes: r.minutes }); });
    add({ kind: 'validate', leg, minutes: validation === null ? null : validation * 60 });
  });

  const legsTotal = legs.length && legs.every(l => l.total !== null) ? sum(legs.map(l => l.total)) : null;
  const total = (legsTotal === null || (vm && vm.minutes === null)) ? null : legsTotal + (vm ? vm.minutes : 0);

  const hitKeys = new Set(dbs.map(d => d.key));
  const appsHit = uniq(dbs.flatMap(d => d.apps)).map(name => {
    const a = applications.get(name);
    const hit = a.dbs.filter(d => hitKeys.has(d.key)).length;
    return { name, hit, of: a.dbs.length, full: hit === a.dbs.length };
  }).sort((a, b) => (b.full - a.full) || byName(a.name, b.name));

  return { kind, spec, dbs, legs, vm, steps, total, clock, appsHit };
}

/* One plain paragraph that says what happened and what it takes, from the run's own numbers. */
function scenarioStory(run, ctx) {
  const n = run.dbs.length, k = run.legs.length, a = run.appsHit.length;
  const full = run.appsHit.filter(x => x.full).length;
  const where = k === 1 ? `on <b>${esc(run.legs[0].inst.instance)}</b> ${envTag(run.legs[0].inst.environment)}` : `across ${k} SQL instances`;
  const subject = n === 1 ? dbName(run.dbs[0]) : `${n} databases`;
  const appNames = nameList(run.appsHit.map(x => `<b>${esc(x.name)}</b>`), 4);
  const affected = a
    ? `${plural(a, 'application')} ${a === 1 ? 'depends' : 'depend'} on ${n === 1 ? 'it' : 'them'}: ${appNames}.`
    : `No application is recorded against ${n === 1 ? 'it' : 'them'}.`;

  const down = `${plural(a, 'application')} ${a === 1 ? 'is' : 'are'} down until then &mdash; ${full} completely, ${a - full} partly: ${appNames}.`;

  switch (run.kind) {
    case 'app-lost': {
      const others = run.appsHit.filter(x => x.name !== ctx.app.name).length;
      const shared = ctx.collateral.length;
      return `<b>${esc(ctx.app.name)}</b> is lost${ctx.filter !== 'ALL' ? ` in ${envTag(ctx.filter)}` : ''}. Its ${plural(n, 'database')} ${where} ${n === 1 ? 'is' : 'are'} restored one at a time, after SQL Server is rebuilt on each instance.`
        + (others ? ` ${plural(others, 'other application')} ${others === 1 ? 'uses' : 'use'} some of the same databases and ${others === 1 ? 'loses' : 'lose'} them too.` : '')
        + (shared ? ` ${plural(shared, 'other application')} ${shared === 1 ? 'shares' : 'share'} these instances but ${shared === 1 ? 'is' : 'are'} not restored here.` : '');
    }
    case 'corrupt':
      return `${subject} ${where} ${n === 1 ? 'is' : 'are'} corrupt. ${n === 1 ? 'It is' : 'Each is'} restored from backup, one at a time; nothing is rebuilt. ${affected}`;
    case 'inst-lost':
      return `<b>${esc(ctx.inst.instance)}</b> ${envTag(ctx.inst.environment)} is lost, but its VM <b>${esc(ctx.inst.server)}</b> survives. SQL Server is rebuilt and all ${plural(n, 'database')} on it restored, one at a time. ${down}`;
    case 'vm-lost': {
      const list = run.legs.map(l => `<b>${esc(l.inst.instance)}</b> ${envTag(l.inst.environment)}`).join(', ');
      return `VM <b>${esc(run.vm.name)}</b> is lost` + (k === 1 ? `. It runs one SQL instance, ${list}.` : `, taking ${k} SQL instances with it: ${list}.`)
        + ` The VM is rebuilt first, then SQL Server and all ${plural(n, 'database')} ${k === 1 ? 'on it' : 'for each instance in turn'}. ${down}`;
    }
  }
  return '';
}

/* Observed facts about one database, read for this particular failure. */
function scenarioFindings(family, d) {
  const out = [];
  const got = wasCollected(d);
  const model = String(d.recoveryModel || '').toUpperCase();

  if (d.state && d.state !== 'ONLINE') out.push(['bad', `Observed state is ${d.state}.`]);
  if (got && !d.lastFullBackup) out.push(['bad', `No full backup in the last ${WINDOW_DAYS} days. There is nothing to restore from.`]);

  if (family === 'corrupt' && d.lastFullBackup) {
    if (model === 'SIMPLE') {
      out.push(['warn', `SIMPLE recovery: it can only go back to the last full backup (${d.lastFullBackup}). Everything after that is lost.`]);
    } else if (model === 'FULL' || model === 'BULK_LOGGED') {
      if (d.lastLogBackup) out.push(['ok', `Can be restored to just before the damage. Log backups run up to ${d.lastLogBackup}.`]);
      else out.push(['warn', `${model} recovery, but no log backup in the last ${WINDOW_DAYS} days. Only the last full backup can be restored.`]);
    }
  }
  if (family === 'corrupt') {
    if (d.lastGoodCheckDb === 'Never') out.push(['bad', 'No clean DBCC CHECKDB on record. Any backup may carry the same corruption.']);
    else if (d.lastGoodCheckDb) out.push(['info', `Last clean CHECKDB ${d.lastGoodCheckDb}. Backups taken after that are not proven clean.`]);
  }

  out.push(MECH_NOTE[family][mechClass(d.mechanism)]);

  if (family === 'corrupt' && d.shared) {
    out.push(['warn', `Used by ${d.apps.length} applications. All of them are down until it is back.`]);
  }
  return out;
}

function scenarioRemedies(run) {
  const fam = run.spec.family;
  const out = [];
  const restores = run.steps.filter(s => s.kind === 'restore' && s.minutes !== null).map(s => s.minutes);
  if (restores.length > 1) {
    out.push(`Restoring the ${restores.length} databases at the same time instead of one after another would bring the restore portion from ${hoursText(sum(restores))} to about ${hoursText(Math.max(...restores))}, the longest single restore &mdash; if the storage and people exist to run them together.`);
  }
  const builds = run.steps.filter(s => s.kind === 'rebuild' && s.minutes !== null).map(s => s.minutes);
  if (builds.length > 1) {
    out.push(`Rebuilding SQL Server on the ${builds.length} instances at the same time would bring that portion from ${hoursText(sum(builds))} to ${hoursText(Math.max(...builds))}.`);
  }
  if (run.vm) {
    out.push('A standby VM takes the VM build off the critical path; a VM template with SQL Server already installed shortens the SQL rebuild on every instance.');
  }
  if (fam === 'lost') {
    const replicas = run.dbs.filter(d => HAS_REPLICA.includes(mechClass(d.mechanism))).length;
    const rest = run.dbs.length - replicas;
    if (replicas) out.push(`${replicas} of these ${run.dbs.length} databases have a replica elsewhere. A tested failover takes them out of the restore path.`);
    if (rest) out.push(`A synchronous replica or warm standby for ${replicas ? `the other ${plural(rest, 'database')}` : (rest === 1 ? 'this database' : `these ${rest} databases`)} takes the restore off the critical path entirely.`);
  }
  if (fam === 'corrupt') {
    const simple = run.dbs.filter(d => String(d.recoveryModel || '').toUpperCase() === 'SIMPLE').length;
    if (simple) out.push(`${plural(simple, 'database')} ${simple === 1 ? 'runs' : 'run'} in SIMPLE recovery. FULL recovery with regular log backups would let ${simple === 1 ? 'it' : 'them'} go back to just before the damage, not only to the last full backup.`);
    const never = run.dbs.filter(d => d.lastGoodCheckDb === 'Never').length;
    if (never) out.push(`${plural(never, 'database')} ${never === 1 ? 'has' : 'have'} no clean DBCC CHECKDB on record. A scheduled integrity check shows which backup is safe to restore.`);
  }
  return out;
}

function scenarioGaps(run, ctx) {
  const out = [];
  const nullDbs = run.steps.filter(s => s.kind === 'restore' && s.minutes === null).map(s => s.db);
  if (nullDbs.length) {
    out.push(`${plural(nullDbs.length, 'database')} with a NULL restore time: ${nameList(nullDbs.map(d => esc(d.database)), 6)}. Run collection (-CollectSqlEvidence) to measure ${nullDbs.length === 1 ? 'it' : 'them'}; until then the real figure is higher.`);
  }
  if (run.vm && run.vm.minutes === null) out.push(`${esc(envBucket(run.vm.environment))} VM build hours are NULL, so the time to recover is NULL. Enter them on the Framework panel.`);
  const nullBuild = uniq(run.steps.filter(s => s.kind === 'rebuild' && s.minutes === null).map(s => envBucket(s.leg.inst.environment)));
  if (nullBuild.length) out.push(`${esc(nullBuild.join(', '))} SQL rebuild hours are NULL, so the time to recover is NULL. Enter them on the Framework panel.`);
  if (run.steps.some(s => s.kind === 'validate' && s.minutes === null)) out.push('Validation hours are NULL and counted as nothing. Enter them on the Framework panel.');
  out.push('Detecting the failure and deciding to act.');
  out.push('Log backups restored on top of the full backup, unless the timed restore already included them.');
  if (run.spec.family === 'lost') out.push('Failing over to a replica. It has never been timed here, so the timeline shows the restore path.');
  if (run.kind === 'app-lost' && ctx.collateral.length) out.push(`${plural(ctx.collateral.length, 'other application')} on the same instances. Their databases are not restored in this scenario.`);
  return out;
}

function scenarioResultHtml(run, ctx) {
  const spec = run.spec, fam = spec.family, dbs = run.dbs;
  const sized = dbs.filter(d => d.sizeGb != null);
  const gb = sized.length ? sum(sized.map(d => d.sizeGb)) : null;
  const full = run.appsHit.filter(a => a.full).length;
  const nullRestores = run.steps.filter(s => s.kind === 'restore' && s.minutes === null).length;
  const nullBuildEnvs = uniq(run.steps.filter(s => s.kind === 'rebuild' && s.minutes === null).map(s => envBucket(s.leg.inst.environment)));
  const emptyLegs = run.legs.filter(l => l.work.total === null).length;

  const totalSub = run.total !== null
    ? (nullRestores ? `Leaves out ${plural(nullRestores, 'NULL restore')}, so the real figure is higher` : 'One step at a time, the longest it takes')
    : (run.vm && run.vm.minutes === null) ? `${envBucket(run.vm.environment)} VM build hours are NULL`
    : (nullBuildEnvs.length ? `${nullBuildEnvs.join(', ')} SQL rebuild hours are NULL` : `${plural(emptyLegs, 'instance')} with no restore figure at all`);

  const kpis = [
    ['Time to recover', fmtH(run.total), esc(totalSub), spec.colour],
    ['Databases restored', dbs.length, spreadHtml(dbs), 'var(--info)'],
    ['Data to restore', fmtGb(gb), gb === null ? 'No size captured yet' : `across ${plural(sized.length, 'sized database')}`, 'var(--accent)'],
    ['Applications affected', run.appsHit.length, run.appsHit.length ? `${full} completely, ${run.appsHit.length - full} partly` : 'None recorded', 'var(--high)']
  ].map(kpiHtml).join('');

  /* The timeline. Each bar starts where the one before it ended. */
  const scale = Math.max(run.clock, 1);
  const pct = m => (m / scale * 100).toFixed(2);
  let stepNo = 0;
  const assumption = (s, source) => ({
    sub: s.minutes === null ? `${esc(source)} are NULL. Enter them on the Framework panel.` : `${esc(source)}, set on the Framework panel`,
    basis: s.minutes === null ? '<span class="badge b-none">NULL</span>' : '<span class="badge b-assume">Assumption</span>'
  });
  const stepRow = s => {
    stepNo++;
    let colour, what, sub, basis;
    if (s.kind === 'restore') {
      const d = s.db;
      colour = BASIS_HEX[s.r.basis];
      what = `${RESTORE_VERB[fam]} ${dbName(d)}${d.shared ? `<span class="shared-tag">used by ${d.apps.length} apps</span>` : ''}`;
      sub = (s.r.basis === 'derived' ? '' : `${esc(gbText(d.sizeGb))} &middot; `) + esc(s.r.detail);
      basis = basisBadge(s.r);
    } else if (s.kind === 'vm') {
      colour = 'var(--med)';
      what = `Build the replacement VM <b>${esc(s.vm.name)}</b>`;
      ({ sub, basis } = assumption(s, `${envBucket(s.vm.environment)} VM build hours`));
    } else if (s.kind === 'rebuild') {
      colour = 'var(--high)';
      what = 'Install and configure SQL Server to the point a restore can start';
      ({ sub, basis } = assumption(s, `${envBucket(s.leg.inst.environment)} SQL rebuild hours`));
    } else {
      colour = 'var(--brand)';
      what = 'Validate, reconnect the applications and hand over';
      ({ sub, basis } = assumption(s, 'Validation hours'));
    }
    const bar = s.minutes === null
      ? `<i class="nullbar" style="left:calc(${pct(s.start)}% - ${(s.start / scale * 12).toFixed(1)}px)"></i>`
      : `<i style="left:${pct(s.start)}%;width:${pct(s.minutes)}%"></i>`;
    return `<tr>
      <td class="num mono">${stepNo}</td>
      <td><span class="step-kind" style="--k-c:${colour}">${STEP_LABEL[s.kind]}</span>${what}<div class="note">${sub}</div></td>
      <td class="num mono"><b>${fmtH(s.minutes)}</b></td>
      <td>${basis}</td>
      <td class="clock">T+ ${clockText(s.end)}${s.open ? ' <span class="nul">+ NULL</span>' : ''}</td>
      <td><div class="track" style="--k-c:${colour}">${bar}</div></td></tr>`;
  };
  const vmRows = run.vm
    ? `<tr class="leg"><td colspan="6">VM ${esc(run.vm.name)} ${envTag(run.vm.environment)} &nbsp;&middot;&nbsp; carries ${plural(run.legs.length, 'SQL instance')}</td></tr>`
      + run.steps.filter(s => s.kind === 'vm').map(stepRow).join('')
    : '';
  const rows = vmRows + run.legs.map(leg => {
    const inst = leg.inst;
    const head = `<tr class="leg"><td colspan="6">${esc(inst.instance)} ${envTag(inst.environment)} &nbsp;&middot;&nbsp; VM ${nz(inst.server)} &nbsp;&middot;&nbsp; ${plural(leg.dbs.length, 'database')} &nbsp;&middot;&nbsp; this instance ${fmtH(leg.total)}</td></tr>`;
    return head + run.steps.filter(s => s.leg === leg).map(stepRow).join('');
  }).join('');
  const counted = run.steps.filter(s => s.minutes !== null).length;

  /* Who is down. */
  const appsHtml = run.appsHit.length
    ? `<div class="apps-hit">${run.appsHit.map(a => `<button type="button" class="app-hit ${a.full ? 'full' : ''}" data-goapp="${esc(a.name)}" title="Open ${esc(a.name)} on the Application Overview"><b>${esc(a.name)}</b><span>${a.full ? `all ${plural(a.of, 'database')}` : `${a.hit} of ${a.of} databases`}</span></button>`).join('')}</div>`
    : '<p class="note">No application is recorded against these databases.</p>';
  const collateralHtml = run.kind === 'app-lost' && ctx.collateral.length
    ? `<p class="note" style="margin-top:9px">Sharing these instances but not restored here: ${esc(nameList(ctx.collateral, 8))}. If an instance itself has gone, run it as an instance scenario: `
      + run.legs.map(l => `<button type="button" class="link" data-goinst="${esc(l.inst.instance)}" data-kind="inst-lost">${esc(l.inst.instance)}</button> ${envTag(l.inst.environment)}`).join(' &nbsp; ') + '</p>'
    : '';

  /* What the evidence says. A note that applies to every row is said once, above the table. */
  const cols = CHECK_COLS[fam];
  const showInst = run.legs.length > 1;
  const perDb = run.legs.flatMap(l => l.dbs).map(d => ({ d, f: scenarioFindings(fam, d) }));
  const fkey = f => f.join('|');
  const common = perDb.length > 1 ? perDb[0].f.filter(f => perDb.every(x => x.f.some(g => fkey(g) === fkey(f)))) : [];
  const commonKeys = new Set(common.map(fkey));
  const verdict = f => `<span class="verdict v-${f[0]}">${esc(f[1])}</span>`;
  const own = perDb.map(x => x.f.filter(f => !commonKeys.has(fkey(f))));
  const showNotes = own.some(list => list.length);
  const checkRows = perDb.map(({ d }, i) => `<tr>
      <td><b>${esc(d.database)}</b> ${envTag(d.environment)}</td>
      ${showInst ? `<td>${esc(d.instance)} ${envTag(instances.get(d.instance).environment)}</td>` : ''}
      ${cols.map(c => `<td class="${CHECK_CELLS[c][2]}">${CHECK_CELLS[c][1](d)}</td>`).join('')}
      ${showNotes ? `<td style="min-width:260px">${own[i].map(verdict).join('')}</td>` : ''}</tr>`).join('');
  const commonHtml = common.length
    ? `<div class="callout" style="margin:0 0 10px;background:#fff"><b>Applies to every database here</b>${common.map(verdict).join('')}</div>` : '';

  const remedies = scenarioRemedies(run);
  const gaps = scenarioGaps(run, ctx);

  return `
    <div class="scn-story" style="--scn-c:${spec.colour}">${scenarioStory(run, ctx)}</div>
    <div class="grid4">${kpis}</div>

    <div class="scn-h">Step by step, one at a time</div>
    <div class="tbl-wrap" style="max-height:440px"><table class="table scn-steps">
      <thead><tr><th class="num">#</th><th>Step</th><th class="num">Duration</th><th>Basis</th><th class="num">Clock after</th><th>Timeline</th></tr></thead>
      <tbody>${rows}</tbody></table></div>
    <p class="note" style="margin-top:7px">${counted} of ${run.steps.length} steps have a figure. The clock adds up only those; <span class="nul">+ NULL</span> means a step with no figure has already been passed.</p>

    <div class="scn-h">Applications affected</div>
    ${appsHtml}${collateralHtml}

    <div class="scn-h">What the evidence says about this failure</div>
    ${commonHtml}
    <div class="tbl-wrap" style="max-height:340px"><table class="table">
      <thead><tr><th>Database</th>${showInst ? '<th>SQL instance</th>' : ''}${cols.map(c => `<th class="${CHECK_CELLS[c][2].includes('num') ? 'num' : ''}">${CHECK_CELLS[c][0]}</th>`).join('')}${showNotes ? '<th>What it means here</th>' : ''}</tr></thead>
      <tbody>${checkRows}</tbody></table></div>

    <div class="grid2" style="margin-top:14px">
      <div class="card" style="--accent-c:var(--accent)"><b class="h">Ways to bring this down</b>
        ${remedies.length ? `<ul class="scn-list">${remedies.map(x => `<li>${x}</li>`).join('')}</ul>` : '<p>Nothing in the collected evidence points to a shorter path.</p>'}</div>
      <div class="card" style="--accent-c:var(--med)"><b class="h">Not counted in this figure</b>
        <ul class="scn-list">${gaps.map(x => `<li>${x}</li>`).join('')}</ul></div>
    </div>`;
}

/* One scenario block: the drop-down, a database picker when the failure is database-level,
   and the run-down. Returns a redraw function so the relationship map can drive it too. */
function scenarioControlsHtml(o) {
  const spec = SCENARIOS[o.state.kind];
  const options = o.kinds.map(k => `<option value="${k}"${k === o.state.kind ? ' selected' : ''}>${esc(SCENARIOS[k].label)}</option>`).join('');
  let picker = '';
  if (!spec.rebuild) {
    const mixed = o.pool.some(d => isProd(d.environment)) && o.pool.some(d => !isProd(d.environment));
    picker = `<div class="scn-picker">
      <div class="ph"><span data-count></span>
        <span class="acts"><button type="button" class="link" data-pick="all">Tick all</button>${mixed ? '<button type="button" class="link" data-pick="prod">Production only</button>' : ''}<button type="button" class="link" data-pick="none">Clear</button></span></div>
      <div class="pb">${groupByInstance(o.pool).map(([inst, list]) =>
        (o.grouped ? `<div class="scn-group">${esc(inst.instance)} ${envTag(inst.environment)}</div>` : '')
        + list.map(d => `<label class="scn-opt"><input type="checkbox" data-key="${esc(d.key)}"${o.state.picked.has(d.key) ? ' checked' : ''}>
            <span>${dbName(d)}${d.shared ? `<span class="shared-tag">used by ${d.apps.length} apps</span>` : ''}</span>
            <span class="f">${esc(gbText(d.sizeGb))}</span><span class="f">${esc(hoursText(restoreOf(d).minutes))}</span></label>`).join('')).join('')}</div>
    </div>`;
  }
  return `<div class="scn-bar"><label>Scenario<select class="search" data-scn aria-label="Scenario">${options}</select></label>
    <div class="scn-what">${esc(spec.what)}</div></div>${picker}`;
}

function mountScenario(o) {
  const drawResult = () => {
    const spec = SCENARIOS[o.state.kind];
    const counter = o.controls.querySelector('[data-count]');
    const ticked = o.pool.filter(d => o.state.picked.has(d.key));
    if (counter) counter.innerHTML = `<b>${ticked.length}</b> of ${o.pool.length} databases ticked${ticked.length ? ' &nbsp;&middot;&nbsp; ' + spreadHtml(ticked) : ''}`;
    const dbs = spec.rebuild ? o.scope(o.state.kind) : ticked;
    if (!dbs.length) {
      o.result.innerHTML = `<div class="empty">${spec.rebuild ? 'Nothing is in scope for this filter.' : 'Tick at least one database to run this scenario.'}</div>`;
    } else {
      o.result.innerHTML = scenarioResultHtml(runScenario(o.state.kind, dbs), o.ctx);
      wireScenarioLinks(o.result);
    }
    if (o.after) o.after();
  };
  const draw = () => {
    const body = o.controls.querySelector('.pb');
    const scroll = body ? body.scrollTop : 0;
    o.controls.innerHTML = scenarioControlsHtml(o);
    const fresh = o.controls.querySelector('.pb');
    if (fresh) fresh.scrollTop = scroll;
    o.controls.querySelector('[data-scn]').onchange = e => { o.state.kind = e.target.value; draw(); };
    o.controls.querySelectorAll('input[data-key]').forEach(cb => cb.onchange = () => {
      if (cb.checked) o.state.picked.add(cb.dataset.key); else o.state.picked.delete(cb.dataset.key);
      drawResult();
    });
    o.controls.querySelectorAll('[data-pick]').forEach(b => b.onclick = () => {
      o.state.picked.clear();
      o.pool.forEach(d => { if (b.dataset.pick === 'all' || (b.dataset.pick === 'prod' && isProd(d.environment))) o.state.picked.add(d.key); });
      draw();
    });
    drawResult();
  };
  draw();
  return draw;
}

function wireScenarioLinks(root) {
  root.querySelectorAll('[data-goapp]').forEach(b => b.onclick = () => {
    selectedApp = b.dataset.goapp;
    document.getElementById('envFilter').value = 'ALL';
    renderAppList(); renderApp(); showView('appView');
  });
  root.querySelectorAll('[data-goinst]').forEach(b => b.onclick = () => openInstance(b.dataset.goinst, b.dataset.kind || null));
}

function renderApp() {
  if (!selectedApp) return;
  const app = applications.get(selectedApp);

  const envs = uniq(app.rows.map(r => r.environment)).sort((a, b) => envRank(a) - envRank(b));
  const current = filter.value;
  filter.innerHTML = '<option value="ALL">All environments</option>' +
    envs.map(e => `<option value="${esc(e)}">${esc(e)}</option>`).join('');
  filter.value = envs.includes(current) ? current : 'ALL';

  const m = applicationMttr(app, filter.value);
  const sized = m.dbs.filter(d => d.sizeGb != null);
  const gb = sized.length ? sum(sized.map(d => d.sizeGb)) : null;

  document.getElementById('appTitle').textContent = app.name;
  document.getElementById('appSubtitle').innerHTML =
    `${envs.map(envTag).join(' ')} &nbsp;&middot;&nbsp; ${m.legs.length} SQL instances &nbsp;&middot;&nbsp; ${m.dbs.length} databases`
    + (m.collateral.length ? ` &nbsp;&middot;&nbsp; shares infrastructure with <b>${m.collateral.length}</b> other application${m.collateral.length === 1 ? '' : 's'}` : '');

  document.getElementById('appKpis').innerHTML = [
    ['Databases in scope', m.dbs.length, m.dbs.length ? spreadHtml(m.dbs) : 'None in this filter', 'var(--info)'],
    ['SQL instances', m.legs.length, m.collateral.length ? `${m.collateral.length} other applications on the same hosts` : 'No other application shares these hosts', 'var(--accent)']
  ].map(kpiHtml).join('');

  const details = applicationDetails.filter(x => x.application === selectedApp);
  const box = document.getElementById('appDetails');
  if (details.length) {
    box.innerHTML = details.map(x => `<div><b>${esc(x.sheet)}</b> &mdash; ` + x.fields
      .map(f => `${esc(f.label)}: <b>${f.isNull ? NUL : esc(f.value)}</b>`).join(' &middot; ') + '</div>').join('');
    box.classList.remove('hidden');
  } else { box.classList.add('hidden'); }

  const quantified = m.mix.measured + m.mix.derived;
  document.getElementById('appNote').innerHTML = m.dbs.length
    ? `${quantified} of ${m.dbs.length} databases have a restore figure behind them`
      + (m.mix.none
          ? `; the other ${m.mix.none} ${m.mix.none === 1 ? 'is' : 'are'} NULL and excluded from the MTTR above, so the real figure is higher until they are collected.`
          : `. Every database in scope is quantified.`)
    : '';

  /* The scenario keeps its ticks while the application and filter stay the same. */
  const pool = groupByInstance(m.dbs).flatMap(([, list]) => list);
  const scope = `${app.name}|${filter.value}`;
  if (appScn.scope !== scope) { appScn.scope = scope; appScn.picked = new Set(defaultPick(pool)); }
  appScn.draw = mountScenario({
    controls: document.getElementById('appScnControls'),
    result: document.getElementById('appScnResult'),
    kinds: APP_SCENARIOS, state: appScn, pool, grouped: true,
    scope: () => pool,
    ctx: { app, filter: filter.value, collateral: m.collateral },
    after: () => renderAppTopology(app, m)
  });
  renderAppEnvironments(app, m, filter.value);
}

/* The relationship map: application, environment, SQL instance, databases. */
function renderAppTopology(app, m) {
  const el = document.getElementById('appTopo');
  const legendEl = document.getElementById('appTopoLegend');

  if (!m.legs.length) {
    el.innerHTML = '<div class="empty">No databases for this filter.</div>';
    legendEl.innerHTML = '';
    return;
  }

  const byEnv = new Map();
  m.legs.forEach(leg => {
    const e = leg.instance.environment;
    if (!byEnv.has(e)) byEnv.set(e, []);
    byEnv.get(e).push(leg);
  });

  /* With a database-level scenario picked, the map is also a picker: ticked databases are
     ringed, the rest recede, and clicking a database adds it or takes it out. */
  const picking = !SCENARIOS[appScn.kind].rebuild;
  el.classList.toggle('scn-pick', picking && appScn.picked.size > 0);

  const cards = [...byEnv.entries()].sort((a, b) => envRank(a[0]) - envRank(b[0])).map(([environment, legs]) => {
    const envDbs = legs.flatMap(l => l.dbs);
    const totals = legs.map(l => l.total).filter(t => t !== null);
    const envTotal = totals.length === legs.length ? sum(totals) : null;

    const stack = legs.slice().sort((a, b) => byName(a.instance.instance, b.instance.instance)).map(leg => {
      const inst = leg.instance;
      const release = versionLabel(inst.meta);
      const chips = leg.dbs.map(d => {
        const r = restoreOf(d);
        const tip = `${d.database} (${d.environment}) on ${d.instance}\n${gbText(d.sizeGb)} \u00b7 restore ${hoursText(r.minutes)} \u00b7 ${r.label}\n${r.detail}`
                  + (picking ? `\nClick to ${appScn.picked.has(d.key) ? 'take it out of' : 'add it to'} the scenario` : '');
        const inner = `<b>${esc(d.database)}</b><span>${esc(hoursText(r.minutes))}</span>`;
        return picking
          ? `<button type="button" class="dbchip ${appScn.picked.has(d.key) ? 'hit' : ''}" data-dbkey="${esc(d.key)}" style="--basis-c:${BASIS_HEX[r.basis]}" title="${esc(tip)}">${inner}</button>`
          : `<span class="dbchip" style="--basis-c:${BASIS_HEX[r.basis]}" title="${esc(tip)}">${inner}</span>`;
      }).join('');

      return `<div class="inst-node">
        <div class="inst-top">
          <div style="min-width:0">
            <button class="inst-link" data-instance="${esc(inst.instance)}">${esc(inst.instance)}</button> ${envTag(inst.environment)}
            <div class="inst-meta">Host ${nz(inst.server)} &middot; ${release ? esc(release) : NUL}${inst.siblings ? ` &middot; ${inst.siblings} other instance${inst.siblings === 1 ? '' : 's'} on this host` : ''}</div>
          </div>
          <div class="inst-fig"><b>${leg.dbs.length}</b> databases<br><b>${fmtH(leg.total)}</b> this leg</div>
        </div>
        <div class="dbchips">${chips}</div>
      </div>`;
    }).join('');

    return `<div class="env-card" style="--env-c:${envHex(environment)}">
      <div class="eh">
        <div class="env-circle">${esc(environment)}</div>
        <div class="env-sum">
          <b>${legs.length} SQL instance${legs.length === 1 ? '' : 's'}</b>
          <span>${envDbs.length} database${envDbs.length === 1 ? '' : 's'} &middot; ${hoursText(envTotal)} to recover this environment</span>
        </div>
      </div>
      <div class="inst-stack">${stack}</div>
    </div>`;
  }).join('');

  el.innerHTML = `<div class="flow-root"><div class="root-circle">
      <b>${esc(app.name)}</b>
      <span>${m.dbs.length} database${m.dbs.length === 1 ? '' : 's'}<br>${m.legs.length} SQL instance${m.legs.length === 1 ? '' : 's'}<br>${esc(spreadText(m.dbs))}</span>
    </div></div>
    <div class="pointer-line"></div>
    <div class="env-grid">${cards}</div>`;

  el.querySelectorAll('[data-instance]').forEach(b => b.onclick = () => openInstance(b.dataset.instance, null));
  el.querySelectorAll('[data-dbkey]').forEach(b => b.onclick = () => {
    const key = b.dataset.dbkey;
    if (appScn.picked.has(key)) appScn.picked.delete(key); else appScn.picked.add(key);
    appScn.draw();
  });

  legendEl.innerHTML =
    envCounts(m.dbs).map(([e, n]) => `<span><i class="sw" style="background:${envHex(e)}"></i>${esc(e)} ${n}</span>`).join('')
    + `<span style="margin-left:14px"><i class="rd" style="background:${BASIS_HEX.measured}"></i>measured</span>`
    + `<span><i class="rd" style="background:${BASIS_HEX.derived}"></i>derived</span>`
    + `<span><i class="rd" style="background:${BASIS_HEX.none}"></i>NULL</span>`
    + (picking ? `<span style="margin-left:14px"><i class="sw" style="background:var(--crit-bg);box-shadow:0 0 0 2px var(--crit)"></i>in the scenario: ${appScn.picked.size}</span>` : '')
    + `<span style="margin-left:auto;color:var(--ink3)">The coloured edge of each database is the basis of its restore time.</span>`;
}

function renderDeck() {
  const allDbs = [...databases.values()];
  const sized = allDbs.filter(d => d.sizeGb != null);
  const totalGb = sized.length ? sum(sized.map(d => d.sizeGb)) : null;
  const withFigure = allDbs.filter(d => restoreOf(d).minutes !== null).length;

  const instList = [...instances.values()].map(i => ({ inst: i, m: instanceMttr(i) }));
  const quantified = instList.filter(x => x.m.complete !== null);
  const worstInstance = quantified.slice().sort((a, b) => b.m.complete - a.m.complete)[0] || null;

  const appList = [...applications.values()].map(a => ({ app: a, m: applicationMttr(a, 'ALL') }));
  const worstApp = appList.filter(x => x.m.complete !== null).sort((a, b) => b.m.complete - a.m.complete)[0] || null;

  const dbTimes = allDbs.map(d => ({ db: d, r: restoreOf(d) })).filter(x => x.r.minutes !== null);
  const worstDb = dbTimes.slice().sort((a, b) => b.r.minutes - a.r.minutes)[0] || null;
  const medianDb = dbTimes.length
    ? dbTimes.slice().sort((a, b) => a.r.minutes - b.r.minutes)[Math.floor(dbTimes.length / 2)] : null;

  document.getElementById('deckLead').textContent =
    `${applications.size} applications run on ${instances.size} SQL instances holding ${databases.size} databases `
    + `(${spreadText(allDbs)}). This is what it currently takes to bring each of those back, one database at a time. `
    + `${withFigure} of ${databases.size} databases have a restore figure behind them; the rest are NULL until they are collected.`;

  document.getElementById('deckKpis').innerHTML = [
    ['Applications in scope', applications.size, `${[...instances.values()].filter(i => i.appList.length > 1).length} SQL instances carry more than one`, 'var(--accent)'],
    ['Databases to recover', databases.size, spreadHtml(allDbs), 'var(--info)'],
    ['Longest single instance loss', worstInstance ? fmtH(worstInstance.m.complete) : NUL,
      worstInstance ? esc(worstInstance.inst.instance) : 'Enter the rebuild hours on the Framework panel', 'var(--crit)'],
    ['Production instances', [...instances.values()].filter(i => isProd(i.environment)).length, `of ${instances.size} total`, 'var(--high)']
  ].map(kpiHtml).join('');

  const scenarios = [
    { h: '1 &middot; A database is lost',
      big: medianDb ? fmtH(medianDb.r.minutes) : NUL,
      who: medianDb
        ? `Typical database. The longest is <b>${esc(worstDb.db.database)}</b> ${envTag(worstDb.db.environment)} at <b>${hoursText(worstDb.r.minutes)}</b>.`
        : 'No database has a restore time yet. Run collection to measure them.' },
    { h: '2 &middot; An application is lost',
      big: worstApp ? fmtH(worstApp.m.complete) : NUL,
      who: worstApp
        ? `Worst case is <b>${esc(worstApp.app.name)}</b>, across ${worstApp.m.legs.length} instance${worstApp.m.legs.length === 1 ? '' : 's'}.`
        : 'Enter the SQL rebuild hours on the Framework panel to quantify this scenario.' },
    { h: '3 &middot; A SQL instance is lost',
      big: worstInstance ? fmtH(worstInstance.m.complete) : NUL,
      who: worstInstance
        ? `<b>${esc(worstInstance.inst.instance)}</b> ${envTag(worstInstance.inst.environment)} &mdash; ${worstInstance.inst.databases.length} databases.`
        : 'Enter the SQL rebuild hours on the Framework panel to quantify this scenario.' }
  ];
  document.getElementById('deckScenarios').innerHTML = scenarios.map(s =>
    `<div class="scenario"><div class="sh">${s.h}</div><div class="sb"><div class="big">${s.big}</div><div class="who">${s.who}</div></div></div>`).join('');
}
