<#
.SYNOPSIS
  PROD vs DR MTTR Intelligence - restore-time simulation for SQL Server (model used: Opus 5.5,
  created by SHYAM KOTTE). Builds one self-contained HTML report from the latest Excel / CSV exports.
.DESCRIPTION
  Run with no parameters: pop-ups ask for the Databases file, the Inventory file and (optional)
  the Restore History file produced by the collector in the report's "Restore History" tab.
  The HTML is saved next to the Databases file and opened in the browser.

  Tabs
    Impact & Restore Timeline - view by server, by Availability Group or by application: what is
                                impacted, the throughput each server actually delivered
                                (min / median / max) with the value used and why, the
                                restore-vs-backup speed setting, restore time per database and in
                                total, restore timeline.
                                Colour: Availability Group / standalone with DR / standalone without DR.
    DR Topology               - one lane per Availability Group: PROD -> DR nodes, resources, edition.
    Restore History           - collector T-SQL + PowerShell runner, upload, match report,
                                timed restore samples.

  Inputs: .xlsx (ImportExcel module or Microsoft Excel) or .csv. Column names are matched
  case-insensitively, ignoring spaces/underscores. Extra columns are ignored; NULL text = empty.
     Databases  (required) ServerName, DatabaseName, DatabaseSizeMB, DatabaseStatus, Environment,
                           ApplicationName, AGName, DatabaseClassification
                (optional) SQLVersion, Platform, SQLEdition / Edition
     Inventory  (required) ServerName
                (optional) LogicalCPUCount, PhysicalMemoryGB, Size, Location, Platform, Edition
     History    (optional) RestoreHistory_All_<date>.xlsx / .csv from the collector (or an export
                           with the same column names). Embedded in the HTML so it opens with it.

  Restore throughput per server, strongest evidence first
     Measured  - timed restores on the server: median (range = min - max)
     Backups   - the server's own backup speed x restore/backup ratio (measured on the estate when
                 timed restores exist, otherwise assumed 75% - adjustable in the report)
     Peers     - servers with similar resources (same VM size > same vCPU and memory > same vCPU >
                 comparable vCPU), their timed restores or backups
     Estate    - estate median (no similar server)
     Assumed   - 200 MB/s until restore history is loaded
     Manual    - any value picked in the report
.EXAMPLE
  .\Build-RestoreBaseline.ps1
.EXAMPLE
  .\Build-RestoreBaseline.ps1 -DatabasesXlsx '.\Current OA Databases.xlsx' -InventoryXlsx '.\SQL OA Inventory.xlsx' -RestoreHistory '.\RestoreHistory_All_20261105_0900.xlsx'
#>

param(
    [string]$DatabasesXlsx,
    [string]$InventoryXlsx,
    [string]$RestoreHistory,
    [string]$OutHtml,
    [string]$DatabasesSheet,
    [string]$InventorySheet,

)

# ---------------------------------------------------------------- file pickers (when not passed as parameters)
function Select-File([string]$Title, [string]$StartDir, [switch]$Optional) {
    Add-Type -AssemblyName System.Windows.Forms
    $dlg = New-Object System.Windows.Forms.OpenFileDialog
    $dlg.Title = $Title
    $dlg.Filter = 'Excel or CSV (*.xlsx;*.xls;*.csv)|*.xlsx;*.xls;*.csv|All files (*.*)|*.*'
    if ($StartDir -and (Test-Path $StartDir)) { $dlg.InitialDirectory = $StartDir }
    $owner = New-Object System.Windows.Forms.Form -Property @{ TopMost = $true }
    try {
        if ($dlg.ShowDialog($owner) -ne [System.Windows.Forms.DialogResult]::OK) {
            if ($Optional) { return $null }
            throw "Cancelled: $Title"
        }
        return $dlg.FileName
    } finally { $owner.Dispose() }
}

$here = if ($PSScriptRoot) { $PSScriptRoot } else { (Get-Location).Path }
$interactive = -not ($DatabasesXlsx -and $InventoryXlsx)
if ($interactive -and $PSCommandPath -and [System.Threading.Thread]::CurrentThread.GetApartmentState() -ne 'STA') {
    # file dialogs need an STA thread - relaunch this script in STA mode with the same parameters
    $exe = (Get-Process -Id $PID).Path
    $argList = @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-STA', '-File', $PSCommandPath)
    foreach ($kv in $PSBoundParameters.GetEnumerator()) {
        if ($kv.Value -is [System.Management.Automation.SwitchParameter]) { if ($kv.Value.IsPresent) { $argList += "-$($kv.Key)" } }
        else { $argList += "-$($kv.Key)"; $argList += [string]$kv.Value }
    }
    & $exe @argList
    return
}
if (-not $DatabasesXlsx) { $DatabasesXlsx = Select-File 'Step 1 of 3: select the DATABASES file (e.g. Current OA Databases.xlsx)' $here }
if (-not $InventoryXlsx) { $InventoryXlsx = Select-File 'Step 2 of 3: select the SERVER INVENTORY file (e.g. SQL OA Inventory.xlsx)' (Split-Path $DatabasesXlsx) }
if ($interactive -and -not $RestoreHistory) {
    $RestoreHistory = Select-File 'Step 3 of 3 (optional): select the RESTORE HISTORY file (RestoreHistory_All_*.xlsx) - Cancel to skip' (Split-Path $DatabasesXlsx) -Optional
}
if (-not $OutHtml) {
    $OutHtml = Join-Path (Split-Path $DatabasesXlsx) ("MTTR_Intelligence_{0:yyyyMMdd_HHmm}.html" -f (Get-Date))
}
Write-Host "Databases file : $DatabasesXlsx"
Write-Host "Inventory file : $InventoryXlsx"
Write-Host ("History file   : {0}" -f $(if ($RestoreHistory) { $RestoreHistory } else { '(none - can be loaded in the report)' }))

<script type="text/plain" id="src-sql">
/* =====================================================================
   MTTR Intelligence - restore & backup throughput collector
   Run on every PROD and DR server (optionally also non-PROD servers:
   their restores, e.g. refreshes, add samples for estimates).
   Read-only: msdb history, SQL Server error log, catalog views.
   SQL Server 2008 R2 -> 2022 and Azure SQL Managed Instance.

   ONE result set, two kinds of rows:
     RowType = 'Database' : one row per user database - latest full restore,
                            latest full backup, backup-throughput statistics,
                            file sizes and instance facts (vCPU, memory, IFI)
     RowType = 'Restore'  : one row per FULL restore whose completion is still
                            in the error log = one measured throughput sample

   Restore duration = msdb restorehistory.restore_date (start) -> first
                      "Database was restored" entry for that database in the
                      error log, before the next restore of the same database
   Restore MB/s     = uncompressed backup size / restore duration
   Backup MB/s      = uncompressed backup size / backup duration - full
                      backups taken on THIS server, VSS snapshots excluded
   Tip: keep 30+ error logs to retain more samples.
   ===================================================================== */
SET NOCOUNT ON;
DECLARE @BackupDays int, @MinSampleMB int;
SET @BackupDays  = 30;     -- window for backup-throughput statistics
SET @MinSampleMB = 1024;   -- smaller backups are left out of the statistics

/* 1. Instance facts */
DECLARE @srv nvarchar(256), @cpu int, @memGB decimal(10,1), @ifi varchar(3), @vmt nvarchar(60), @ver nvarchar(128), @eng int, @ed nvarchar(128);
SET @srv = CAST(SERVERPROPERTY('ServerName') AS nvarchar(256));
SET @ver = CAST(SERVERPROPERTY('ProductVersion') AS nvarchar(128));
SET @eng = CAST(SERVERPROPERTY('EngineEdition') AS int);
SET @ed  = CAST(SERVERPROPERTY('Edition') AS nvarchar(128));
BEGIN TRY SELECT @cpu = cpu_count FROM sys.dm_os_sys_info; END TRY BEGIN CATCH END CATCH;
BEGIN TRY
  EXEC sp_executesql N'SELECT @m = CAST(physical_memory_kb / 1048576.0 AS decimal(10,1)) FROM sys.dm_os_sys_info',
       N'@m decimal(10,1) OUTPUT', @m = @memGB OUTPUT;
END TRY
BEGIN CATCH
  BEGIN TRY
    EXEC sp_executesql N'SELECT @m = CAST(physical_memory_in_bytes / 1073741824.0 AS decimal(10,1)) FROM sys.dm_os_sys_info',
         N'@m decimal(10,1) OUTPUT', @m = @memGB OUTPUT;
  END TRY BEGIN CATCH END CATCH;
END CATCH;
BEGIN TRY
  EXEC sp_executesql N'SELECT @o = MAX(instant_file_initialization_enabled) FROM sys.dm_server_services WHERE filename LIKE N''%sqlservr.exe%''',
       N'@o varchar(3) OUTPUT', @o = @ifi OUTPUT;
END TRY BEGIN CATCH END CATCH;
SET @ifi = CASE @ifi WHEN 'Y' THEN 'Yes' WHEN 'N' THEN 'No' ELSE NULL END;
BEGIN TRY
  EXEC sp_executesql N'SELECT @o = virtual_machine_type_desc FROM sys.dm_os_sys_info',
       N'@o nvarchar(60) OUTPUT', @o = @vmt OUTPUT;
END TRY BEGIN CATCH END CATCH;

/* 2. "Database was restored" entries from every error log */
IF OBJECT_ID('tempdb..#logs') IS NOT NULL DROP TABLE #logs;
CREATE TABLE #logs (ArchiveNo int, LogDate nvarchar(50), LogSizeBytes bigint);
BEGIN TRY INSERT #logs EXEC master.sys.xp_enumerrorlogs 1; END TRY BEGIN CATCH END CATCH;
DECLARE @maxLog int, @i int;
SELECT @maxLog = ISNULL(MAX(ArchiveNo), 6) FROM #logs;
IF OBJECT_ID('tempdb..#el') IS NOT NULL DROP TABLE #el;
CREATE TABLE #el (LogDate datetime, ProcessInfo nvarchar(64), [Text] nvarchar(max));
SET @i = 0;
WHILE @i <= @maxLog
BEGIN
  BEGIN TRY INSERT #el EXEC master.dbo.xp_readerrorlog @i, 1, N'Database was restored'; END TRY BEGIN CATCH END CATCH;
  SET @i = @i + 1;
END;

/* 3. FULL restores, with the completion time when it is in the error log */
IF OBJECT_ID('tempdb..#rs') IS NOT NULL DROP TABLE #rs;
SELECT rh.restore_history_id,
       DatabaseName = rh.destination_database_name,
       StartTime    = rh.restore_date,
       RestoredBy   = rh.user_name,
       FromServer   = bs.server_name,
       FromDatabase = bs.database_name,
       BackupMB     = CAST(bs.backup_size / 1048576.0 AS decimal(18,1)),
       CompressedMB = CAST(bs.compressed_backup_size / 1048576.0 AS decimal(18,1)),
       SourceSec    = DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date),
       DeviceType   = (SELECT MAX(f.device_type) FROM msdb.dbo.backupmediafamily f WHERE f.media_set_id = bs.media_set_id),
       Files        = (SELECT COUNT(*) FROM msdb.dbo.backupmediafamily f WHERE f.media_set_id = bs.media_set_id AND ISNULL(f.mirror, 0) = 0),
       CompletedAt  = lc.CompletedAt,
       Sec          = DATEDIFF(SECOND, rh.restore_date, lc.CompletedAt)
INTO #rs
FROM msdb.dbo.restorehistory rh
LEFT JOIN msdb.dbo.backupset bs ON bs.backup_set_id = rh.backup_set_id
OUTER APPLY (SELECT TOP 1 nx.restore_date AS NextStart
             FROM msdb.dbo.restorehistory nx
             WHERE nx.destination_database_name = rh.destination_database_name
               AND nx.restore_date > rh.restore_date
             ORDER BY nx.restore_date) n
OUTER APPLY (SELECT TOP 1 c.LogDate AS CompletedAt
             FROM #el c
             WHERE CHARINDEX((N'Database: ' + rh.destination_database_name + N',') COLLATE DATABASE_DEFAULT, c.[Text] COLLATE DATABASE_DEFAULT) > 0
               AND c.LogDate >= rh.restore_date
               AND c.LogDate <= ISNULL(n.NextStart, DATEADD(DAY, 3, rh.restore_date))
             ORDER BY c.LogDate) lc
WHERE rh.restore_type = 'D';

IF OBJECT_ID('tempdb..#lr') IS NOT NULL DROP TABLE #lr;
SELECT *, rn = ROW_NUMBER() OVER (PARTITION BY DatabaseName ORDER BY StartTime DESC, restore_history_id DESC)
INTO #lr FROM #rs;
DELETE FROM #lr WHERE rn > 1;

/* 4. Full backups taken on THIS server: latest per database + throughput statistics */
IF OBJECT_ID('tempdb..#lb') IS NOT NULL DROP TABLE #lb;
SELECT database_name, backup_start_date, backup_finish_date, backup_size, compressed_backup_size, media_set_id,
       rn = ROW_NUMBER() OVER (PARTITION BY database_name ORDER BY is_copy_only, backup_finish_date DESC)
INTO #lb
FROM msdb.dbo.backupset
WHERE type = 'D'
  AND ISNULL(is_snapshot, 0) = 0
  AND server_name IN (@@SERVERNAME, @srv);
DELETE FROM #lb WHERE rn > 1;

IF OBJECT_ID('tempdb..#bk') IS NOT NULL DROP TABLE #bk;
SELECT database_name,
       MBps = CAST(backup_size / 1048576.0 / NULLIF(DATEDIFF(SECOND, backup_start_date, backup_finish_date), 0) AS decimal(18,1))
INTO #bk
FROM msdb.dbo.backupset
WHERE type = 'D'
  AND ISNULL(is_snapshot, 0) = 0
  AND server_name IN (@@SERVERNAME, @srv)
  AND backup_finish_date >= DATEADD(DAY, -@BackupDays, GETDATE())
  AND backup_size >= @MinSampleMB * 1048576.0
  AND DATEDIFF(SECOND, backup_start_date, backup_finish_date) >= 10;

IF OBJECT_ID('tempdb..#bks') IS NOT NULL DROP TABLE #bks;
SELECT database_name,
       BackupSamples    = MAX(cnt),
       BackupMBpsMin    = MIN(MBps),
       BackupMBpsMedian = CAST(SUM(CASE WHEN rn IN ((cnt + 1) / 2, (cnt + 2) / 2) THEN MBps ELSE 0 END)
                             / SUM(CASE WHEN rn IN ((cnt + 1) / 2, (cnt + 2) / 2) THEN 1 ELSE 0 END) AS decimal(18,1)),
       BackupMBpsMax    = MAX(MBps)
INTO #bks
FROM (SELECT database_name, MBps,
             rn  = ROW_NUMBER() OVER (PARTITION BY database_name ORDER BY MBps),
             cnt = COUNT(*) OVER (PARTITION BY database_name)
      FROM #bk WHERE MBps IS NOT NULL) s
GROUP BY database_name;

/* 5. Output */
IF OBJECT_ID('tempdb..#out') IS NOT NULL DROP TABLE #out;
CREATE TABLE #out (
  RowType varchar(10), ServerName nvarchar(256), DatabaseName nvarchar(256),
  SizeMB bigint, DataMB bigint, LogMB bigint, [State] nvarchar(60), RecoveryModel nvarchar(60),
  LastRestoreDate varchar(30), RestoredFromServer nvarchar(256), RestoredFromDatabase nvarchar(256), RestoredBy nvarchar(256),
  RestoreBackupMB decimal(18,1), RestoreCompressedMB decimal(18,1), RestoreStartTime varchar(19), RestoreCompletedTime varchar(19),
  RestoreDurationSec int, RestoreDuration varchar(16), RestoreDurationSource varchar(30), RestoreMBps decimal(18,1),
  RestoreDevice varchar(30), RestoreFiles int, SourceBackupSec int, SourceBackupMBps decimal(18,1),
  RestoreCount int, MeasuredRestores int,
  LastFullBackupDate varchar(19), LastFullBackupMB decimal(18,1), LastFullBackupCompressedMB decimal(18,1), LastFullBackupSec int,
  BackupMBps decimal(18,1), BackupDevice varchar(30), BackupFiles int,
  BackupSamples int, BackupMBpsMin decimal(18,1), BackupMBpsMedian decimal(18,1), BackupMBpsMax decimal(18,1),
  HostCPU int, HostMemoryGB decimal(10,1), InstantFileInit varchar(3), VmType nvarchar(60), SqlVersion nvarchar(128), EngineEdition int, SqlEdition nvarchar(128));

INSERT #out (RowType, ServerName, DatabaseName, SizeMB, DataMB, LogMB, [State], RecoveryModel,
             LastRestoreDate, RestoredFromServer, RestoredFromDatabase, RestoredBy, RestoreBackupMB, RestoreCompressedMB,
             RestoreStartTime, RestoreCompletedTime, RestoreDurationSec, RestoreDuration, RestoreDurationSource, RestoreMBps,
             RestoreDevice, RestoreFiles, SourceBackupSec, SourceBackupMBps, RestoreCount, MeasuredRestores,
             LastFullBackupDate, LastFullBackupMB, LastFullBackupCompressedMB, LastFullBackupSec, BackupMBps, BackupDevice, BackupFiles,
             BackupSamples, BackupMBpsMin, BackupMBpsMedian, BackupMBpsMax,
             HostCPU, HostMemoryGB, InstantFileInit, VmType, SqlVersion, EngineEdition, SqlEdition)
SELECT 'Database', @srv, d.name, sz.SizeMB, sz.DataMB, sz.LogMB, d.state_desc, d.recovery_model_desc,
       ISNULL(CONVERT(varchar(19), r.StartTime, 120), 'No restore history'), r.FromServer, r.FromDatabase, r.RestoredBy,
       r.BackupMB, r.CompressedMB, CONVERT(varchar(19), r.StartTime, 120), CONVERT(varchar(19), r.CompletedAt, 120), r.Sec,
       CASE WHEN r.Sec IS NULL THEN NULL
            ELSE CAST(r.Sec / 3600 AS varchar(10)) + ':' + RIGHT('0' + CAST((r.Sec % 3600) / 60 AS varchar(2)), 2)
                 + ':' + RIGHT('0' + CAST(r.Sec % 60 AS varchar(2)), 2) END,
       CASE WHEN r.StartTime IS NULL THEN 'No restore history' WHEN r.CompletedAt IS NOT NULL THEN 'ErrorLog' ELSE 'Not in error log' END,
       CAST(r.BackupMB / NULLIF(r.Sec, 0) AS decimal(18,1)),
       CASE r.DeviceType WHEN 2 THEN 'Disk' WHEN 102 THEN 'Disk' WHEN 9 THEN 'URL' WHEN 7 THEN 'Virtual device'
                         WHEN 5 THEN 'Tape' WHEN 105 THEN 'Tape' ELSE CAST(r.DeviceType AS varchar(10)) END,
       r.Files, r.SourceSec, CAST(r.BackupMB / NULLIF(r.SourceSec, 0) AS decimal(18,1)),
       rc.N, rc.M,
       CONVERT(varchar(19), b.backup_finish_date, 120),
       CAST(b.backup_size / 1048576.0 AS decimal(18,1)),
       CAST(b.compressed_backup_size / 1048576.0 AS decimal(18,1)),
       DATEDIFF(SECOND, b.backup_start_date, b.backup_finish_date),
       CAST(b.backup_size / 1048576.0 / NULLIF(DATEDIFF(SECOND, b.backup_start_date, b.backup_finish_date), 0) AS decimal(18,1)),
       CASE bd.DeviceType WHEN 2 THEN 'Disk' WHEN 102 THEN 'Disk' WHEN 9 THEN 'URL' WHEN 7 THEN 'Virtual device'
                          WHEN 5 THEN 'Tape' WHEN 105 THEN 'Tape' ELSE CAST(bd.DeviceType AS varchar(10)) END,
       bd.Files,
       s.BackupSamples, s.BackupMBpsMin, s.BackupMBpsMedian, s.BackupMBpsMax,
       @cpu, @memGB, @ifi, @vmt, @ver, @eng, @ed
FROM sys.databases d
OUTER APPLY (SELECT SizeMB = SUM(CAST(mf.size AS bigint)) * 8 / 1024,
                    DataMB = SUM(CASE WHEN mf.type = 1 THEN 0 ELSE CAST(mf.size AS bigint) END) * 8 / 1024,
                    LogMB  = SUM(CASE WHEN mf.type = 1 THEN CAST(mf.size AS bigint) ELSE 0 END) * 8 / 1024
             FROM sys.master_files mf WHERE mf.database_id = d.database_id) sz
LEFT JOIN #lr r  ON r.DatabaseName  COLLATE DATABASE_DEFAULT = d.name COLLATE DATABASE_DEFAULT
LEFT JOIN #lb b  ON b.database_name COLLATE DATABASE_DEFAULT = d.name COLLATE DATABASE_DEFAULT
LEFT JOIN #bks s ON s.database_name COLLATE DATABASE_DEFAULT = d.name COLLATE DATABASE_DEFAULT
OUTER APPLY (SELECT N = COUNT(*), M = SUM(CASE WHEN x.CompletedAt IS NOT NULL THEN 1 ELSE 0 END)
             FROM #rs x WHERE x.DatabaseName COLLATE DATABASE_DEFAULT = d.name COLLATE DATABASE_DEFAULT) rc
OUTER APPLY (SELECT DeviceType = MAX(f.device_type), Files = SUM(CASE WHEN ISNULL(f.mirror, 0) = 0 THEN 1 ELSE 0 END)
             FROM msdb.dbo.backupmediafamily f WHERE f.media_set_id = b.media_set_id) bd
WHERE d.database_id > 4                 -- skip master, tempdb, model, msdb
  AND d.source_database_id IS NULL;     -- skip database snapshots

INSERT #out (RowType, ServerName, DatabaseName, LastRestoreDate, RestoredFromServer, RestoredFromDatabase, RestoredBy,
             RestoreBackupMB, RestoreCompressedMB, RestoreStartTime, RestoreCompletedTime, RestoreDurationSec, RestoreDuration,
             RestoreDurationSource, RestoreMBps, RestoreDevice, RestoreFiles, SourceBackupSec, SourceBackupMBps,
             HostCPU, HostMemoryGB, InstantFileInit, VmType, SqlVersion, EngineEdition, SqlEdition)
SELECT 'Restore', @srv, x.DatabaseName, CONVERT(varchar(19), x.StartTime, 120), x.FromServer, x.FromDatabase, x.RestoredBy,
       x.BackupMB, x.CompressedMB, CONVERT(varchar(19), x.StartTime, 120), CONVERT(varchar(19), x.CompletedAt, 120), x.Sec,
       CAST(x.Sec / 3600 AS varchar(10)) + ':' + RIGHT('0' + CAST((x.Sec % 3600) / 60 AS varchar(2)), 2)
         + ':' + RIGHT('0' + CAST(x.Sec % 60 AS varchar(2)), 2),
       'ErrorLog', CAST(x.BackupMB / NULLIF(x.Sec, 0) AS decimal(18,1)),
       CASE x.DeviceType WHEN 2 THEN 'Disk' WHEN 102 THEN 'Disk' WHEN 9 THEN 'URL' WHEN 7 THEN 'Virtual device'
                         WHEN 5 THEN 'Tape' WHEN 105 THEN 'Tape' ELSE CAST(x.DeviceType AS varchar(10)) END,
       x.Files, x.SourceSec, CAST(x.BackupMB / NULLIF(x.SourceSec, 0) AS decimal(18,1)),
       @cpu, @memGB, @ifi, @vmt, @ver, @eng, @ed
FROM #rs x
WHERE x.Sec > 0;

SELECT * FROM #out ORDER BY RowType, DatabaseName, RestoreStartTime;
</script>
<script type="text/plain" id="src-ps">
<# MTTR Intelligence - restore & backup history runner
   Runs Restore_History_All.sql on every server below and writes ONE file for the report.
   Needs : SqlServer module (Invoke-Sqlcmd)  ->  Install-Module SqlServer -Scope CurrentUser
   Output: RestoreHistory_All_<date>.xlsx with the ImportExcel module, else via Microsoft Excel,
           else .csv - the report accepts .xlsx and .csv.
   Azure SQL MI is not listed: its restores are platform-managed.
   Comment out (#) any server you do not want to query. #>
$servers = @(
__SERVERLIST__
)

function tierTxt(g, s) {   // the resource match in words, e.g. "identical VM size Standard_E48ds_v4"
  const me = spec(s);
  return g.tier === 1 ? `identical VM size ${me.vm}` : g.tier === 2 ? `${me.cpu} vCPU and similar memory (${me.mem} GB ±25%)` : g.tier === 3 ? `${me.cpu} vCPU` : `comparable compute (${Math.ceil(me.cpu / 2)}–${me.cpu * 2} vCPU)`;
}
// fewer than 3 samples: widen the range to at least the estate's interquartile spread around the planning value
function band(plan, n, lo, hi) {
  const ref = EST && EST.n >= 5 ? EST : ESTB && ESTB.n >= 5 ? ESTB : null;
  if (n >= 3 || !ref) return { lo, hi, thin: false };
  const l = plan * ref.p25 / ref.med, h = plan * ref.p75 / ref.med;
  return { lo: Math.min(lo, l), hi: Math.max(hi, h), thin: l < lo - 1e-9 || h > hi + 1e-9 };
}
// evidence order: timed restores on the server > its own backups > similar servers > estate > 200 MB/s placeholder
function choose(s) {
  const e = PROF0[s] || { sm: [], own: null, small: 0, bk: null }, kr = kRat();
  const P = { s, sm: e.sm, own: e.own, small: e.small, bk: e.bk, peer: peerGroup(s), bkEst: null, warn: '' };
  if (e.bk) P.bkEst = { k: kr.k, src: kr.src, min: kr.lo * e.bk.min, med: kr.k * e.bk.med, max: kr.hi * e.bk.max };
  let a;
  if (e.own) { const b = band(e.own.med, e.own.n, e.own.min, e.own.max); a = { basis: 'server', plan: e.own.med, lo: b.lo, hi: b.hi, thin: b.thin, n: e.own.n, conf: e.own.n >= 3 ? 'High' : 'Medium' }; }
  else if (P.bkEst) { const b = band(P.bkEst.med, e.bk.n, P.bkEst.min, P.bkEst.max); a = { basis: 'backup', plan: P.bkEst.med, lo: b.lo, hi: b.hi, thin: b.thin, n: e.bk.n, conf: kr.src === 'measured' ? 'Medium' : 'Low' }; }
  else if (P.peer) { const st = P.peer.st, b = band(st.med, st.n, st.min, st.max); a = { basis: 'peer', plan: st.med, lo: b.lo, hi: b.hi, thin: b.thin, n: st.n, conf: P.peer.tier <= 2 && P.peer.kind === 'restore' ? 'Medium' : 'Low' }; }
  else if (EST) a = { basis: 'estate', est: 'restore', plan: EST.med, lo: EST.p25, hi: EST.p75, conf: 'Low' };
  else if (ESTB) a = { basis: 'estate', est: 'backup', plan: kr.k * ESTB.med, lo: kr.lo * ESTB.p25, hi: kr.hi * ESTB.p75, conf: 'Low' };
  else a = { basis: 'assumed', plan: DEF, lo: DEF, hi: DEF, conf: 'None' };
  P.auto = a;
  if (a.basis === 'backup' && P.peer && P.peer.kind === 'restore' && P.peer.tier <= 2) { const r = P.peer.st.med / P.bkEst.med; if (r > 2 || r < 0.5) P.warn = `This server's backup-based estimate (${fmtRate(P.bkEst.med)}) and timed restores on similar servers (${fmtRate(P.peer.st.med)}) differ by ${(r > 1 ? r : 1 / r).toFixed(1)}× — confirm with a test restore.`; }
  Object.assign(P, MAN[s] != null ? { basis: 'manual', plan: MAN[s], lo: MAN[s], hi: MAN[s], conf: 'Manual' } : a);
  return P;
}
const prof = s => PROF[s] || (PROF[s] = choose(s));

/* ============================================================ restore time model */
function vol(x) {
  let bk = null, log = null;
  if (H) [...x.srvs, ...x.drs].forEach(s => {
    const o = H.db[s + '|' + x.db.toLowerCase()]; if (!o) return;
    const mb = num(o.LastFullBackupMB), lg = num(o.LogMB);
    if (lg > 0 && x.srvs.includes(s)) log = Math.max(log || 0, lg);
    if (mb > 0) { const dt = o.LastFullBackupDate || ''; if (!bk || dt > bk.dt) bk = { mb, dt, s, cmb: num(o.LastFullBackupCompressedMB) }; }
  });
  return S.basis === 'backup' && bk ? { V: bk.mb, src: 'backup', bk, log } : { V: x.mb, src: 'alloc', bk, log };
}
function dbTime(x, side, pref) {
  const s = side === 'P' ? (x.srvs.includes(pref) ? pref : x.srvs[0]) : (x.drs.includes(pref) ? pref : x.dr);
  return s ? dbTimeOn(x, s, side) : null;
}
// restore of database x on server s, whether or not s holds a copy today ("what if we restore it on this replica")
function dbTimeOn(x, s, side) {
  const P = prof(s), v = vol(x);
  const own = MAN[s] == null && H ? (H.ev[s] || []).filter(e => e.db.toLowerCase() === x.db.toLowerCase()) : [];
  let o;
  if (own.length) { const st = stats(own.map(e => e.rate)), b = band(st.med, st.n, st.min, st.max); o = { basis: 'actual', rate: st.med, hiR: b.hi, loR: b.lo, conf: 'High', own, st }; }
  else o = { basis: P.basis, rate: P.plan, hiR: P.hi, loR: P.lo, conf: P.conf };
  return Object.assign(o, { x, side, s, P, v, t: v.V / o.rate / 60, best: v.V / o.hiR / 60, worst: v.V / o.loR / 60 });
}
function plan(items) {
  const all = items.filter(Boolean), L = S.excl ? all.filter(it => !it.x.copy) : all;
  const k = Math.max(1, S.streams | 0);
  const sorted = [...L].sort(S.order === 'spt' ? (a, b) => a.t - b.t : (a, b) => b.t - a.t);
  // per-server ceiling: restores running on the same server together never exceed the top of its throughput range
  const cap = new Map();
  L.forEach(it => { const P = prof(it.s); cap.set(it.s, Math.max(cap.get(it.s) || 0, P.hi || 0, (P.auto && P.auto.hi) || 0, it.hiR || 0, it.rate || 0)); });
  const RK = { t: 'rate', best: 'hiR', worst: 'loR' };
  // event simulation: each stream takes the next database when it frees up; the overhead runs at full speed;
  // work is in minutes at full single-restore speed, slowed only while the server's ceiling is exceeded
  const run = key => {
    const queue = sorted.slice(), active = [], seq = [], ovh = S.ovh, rk = RK[key];
    let t = 0, guard = 0;
    const next = lane => { const it = queue.shift(); if (it) active.push({ it, lane, start: t, ph: ovh > 0 ? 'o' : 'r', rem: ovh > 0 ? ovh : it[key] }); };
    for (let i = 0; i < k; i++) next(i);
    while (active.length && guard++ < 1e6) {
      const dem = new Map(); active.forEach(a => { if (a.ph === 'r') dem.set(a.it.s, (dem.get(a.it.s) || 0) + (a.it[rk] || 0)); });
      const phi = active.map(a => { if (a.ph !== 'r') return 1; const d = dem.get(a.it.s), c = cap.get(a.it.s); return c > 0 && d > c ? c / d : 1; });
      let dt = Infinity;
      active.forEach((a, i) => { dt = Math.min(dt, a.ph === 'o' ? a.rem : a.rem / phi[i]); });
      if (!isFinite(dt)) break;
      t += dt;
      for (let i = active.length - 1; i >= 0; i--) {
        const a = active[i];
        a.rem -= a.ph === 'o' ? dt : dt * phi[i];
        if (a.rem > 1e-9) continue;
        if (a.ph === 'o' && a.it[key] > 1e-9) { a.ph = 'r'; a.rem = a.it[key]; continue; }
        active.splice(i, 1); seq.push({ it: a.it, lane: a.lane, start: a.start, end: t }); next(a.lane);
      }
    }
    return { total: t, seq };
  };
  const mix = {}; L.forEach(it => { const kk = KIND(it.basis); mix[kk] = (mix[kk] || 0) + 1; });
  return { n: L.length, excluded: all.length - L.length, k, cap, typ: run('t'), best: run('best'), worst: run('worst'), mix, vol: L.reduce((a, it) => a + it.v.V, 0) };
}

// every database whose restore target is server s (PROD copy on s, or DR copy on s), restored one after another
function srvLoad(s) {
  const L = [];

  return { n: I.length, vol: sum(I, o => o.v.V), t: sum(I, o => o.t), best: sum(I, o => o.best), worst: sum(I, o => o.worst) };
}

function recPath(L, s, isDR, isAG) {
  if (isDR) return { cls: 'warn', txt: 'PROD unaffected — DR copy lost until this server is rebuilt' };
  if (isAG) {
    const others = [...new Set(L.flatMap(r => r.x.srvs))].filter(n => n !== s), drs = [...new Set(L.flatMap(r => r.x.drs))];
    if (others.length) return { cls: 'ok', txt: `AG failover to ${others.join(', ')} — no restore needed` };
    if (drs.length) return { cls: 'ok', txt: `AG failover to DR replica ${drs.join(', ')} — no restore; data loss = replica lag` };
    return { cls: 'off', txt: 'Single-replica AG — restore from backup' };
  }
  const w = L.filter(r => r.x.dr);
  if (w.length === L.length) { const d = [...new Set(w.map(r => r.x.dr))].join(', '); return { cls: 'ok', txt: w.some(r => /failover group/i.test(r.x.mech)) ? `Failover group to ${d}` : `Recover on DR ${d} — ${w[0].x.mech}` }; }
  if (w.length) return { cls: 'warn', txt: `${w.length} of ${L.length} DBs have a DR copy — the rest need a restore from backup` };
  return { cls: 'off', txt: 'No DR copy in inventory — restore from backup' };
}

function parseCSV(t) {
  const first = t.slice(0, t.indexOf('\n') + 1 || t.length), dl = (first.match(/;/g) || []).length > (first.match(/,/g) || []).length ? ';' : (first.match(/\t/g) || []).length > (first.match(/,/g) || []).length ? '\t' : ',';
  const rows = []; let r = [], c = '', q = 0;
  for (let i = 0; i < t.length; i++) {
    const ch = t[i];
    if (q) { if (ch === '"') { if (t[i + 1] === '"') { c += '"'; i++; } else q = 0; } else c += ch; }
    else if (ch === '"') q = 1; else if (ch === dl) { r.push(c); c = ''; }
    else if (ch === '\n' || ch === '\r') { if (ch === '\r' && t[i + 1] === '\n') i++; r.push(c); rows.push(r); r = []; c = ''; }
    else c += ch;
  }
  if (c || r.length) { r.push(c); rows.push(r); }
  return rows;
}
async function unzip(buf) {
  const dv = new DataView(buf); let e = buf.byteLength - 22;
  while (e >= 0 && dv.getUint32(e, true) !== 0x06054b50) e--;
  if (e < 0) throw new Error('not a valid .xlsx file');
  const n = dv.getUint16(e + 10, true); let p = dv.getUint32(e + 16, true); const files = {}, td = new TextDecoder();
  for (let i = 0; i < n; i++) {
    const meth = dv.getUint16(p + 10, true), csz = dv.getUint32(p + 20, true), nl = dv.getUint16(p + 28, true), el = dv.getUint16(p + 30, true), cl = dv.getUint16(p + 32, true), off = dv.getUint32(p + 42, true);
    files[td.decode(new Uint8Array(buf, p + 46, nl))] = { meth, csz, off }; p += 46 + nl + el + cl;
  }
  const get = async name => {
    const f = files[name]; if (!f) return null;
    const lnl = dv.getUint16(f.off + 26, true), lel = dv.getUint16(f.off + 28, true), data = new Uint8Array(buf, f.off + 30 + lnl + lel, f.csz);
    if (f.meth === 0) return td.decode(data);
    const st = new Blob([data]).stream().pipeThrough(new DecompressionStream('deflate-raw')); return await new Response(st).text();
  };
  return { get, names: Object.keys(files) };
}

async function readXlsx(buf) {
  const z = await unzip(buf), q = (el, n) => el.getElementsByTagNameNS('*', n), P = x => new DOMParser().parseFromString(x, 'application/xml');
  const ss = [], sx = await z.get('xl/sharedStrings.xml');
  if (sx) for (const si of q(P(sx), 'si')) ss.push([...q(si, 't')].map(x => x.textContent).join(''));
  let sheet = null; const wb = await z.get('xl/workbook.xml'), rel = await z.get('xl/_rels/workbook.xml.rels');
  if (wb && rel) {
    const sheets = [...q(P(wb), 'sheet')], pick = sheets.find(s => /restore/i.test(s.getAttribute('name') || '')) || sheets[0];
    const rid = pick && (pick.getAttributeNS('http://schemas.openxmlformats.org/officeDocument/2006/relationships', 'id') || pick.getAttribute('r:id'));
    for (const r of q(P(rel), 'Relationship')) if (r.getAttribute('Id') === rid) { const tg = r.getAttribute('Target'); sheet = tg.startsWith('/') ? tg.slice(1) : 'xl/' + tg.replace(/^\.\//, ''); }
  }
  if (!sheet || !z.names.includes(sheet)) sheet = z.names.filter(n => /^xl\/worksheets\/sheet\d+\.xml$/.test(n)).sort((a, b) => parseInt(a.match(/\d+/)[0]) - parseInt(b.match(/\d+/)[0]))[0];
  if (!sheet) throw new Error('no worksheet found');
  const rows = [];
  for (const r of q(P(await z.get(sheet)), 'row')) {
    const a = []; let ci = 0;
    for (const c of q(r, 'c')) {
      const ref = c.getAttribute('r'); if (ref) ci = colIdx(ref);
      const ty = c.getAttribute('t'), v = q(c, 'v')[0];

    }
    rows.push(Array.from(a, x => x ?? ''));
  }
  return rows;
}

function To-Cell($v) {
    if ($null -eq $v -or $v -is [DBNull]) { return '' }
    if ($v -is [datetime]) { return $v.ToString('yyyy-MM-dd HH:mm:ss', [Globalization.CultureInfo]::InvariantCulture) }
    if ($v -is [double] -or $v -is [single] -or $v -is [decimal]) { return ([double]$v).ToString('R', [Globalization.CultureInfo]::InvariantCulture) }
    $s = ([string]$v).Trim()
    if ($s -eq 'NULL') { return '' }   # SSMS "Results to grid" exports write NULL as text
    return $s
}

function Read-Table {
    # Returns @{ Headers = string[]; Rows = List[object[]] } from .csv or .xlsx/.xls (ImportExcel, else Excel COM)
    param([string]$Path, [string]$Sheet, [string]$PreferSheet)
    $full = (Resolve-Path -LiteralPath $Path).Path
    Write-Host "Reading $full" -ForegroundColor Cyan
    $rows = New-Object System.Collections.Generic.List[object]
    $headers = @()
    if ($full -match '\.csv$') {
        $data = @(Import-Csv -LiteralPath $full -Encoding UTF8)
        if ($data.Count -eq 0) { throw "No data in $full" }
        $headers = @($data[0].PSObject.Properties | ForEach-Object { $_.Name })
        foreach ($d in $data) { $rows.Add(@(foreach ($h in $headers) { $d.$h })) }
    }
    elseif (Get-Module -ListAvailable -Name ImportExcel) {
        Import-Module ImportExcel
        if (-not $Sheet -and $PreferSheet) {
            $Sheet = @(Get-ExcelSheetInfo -Path $full | ForEach-Object { $_.Name } | Where-Object { $_ -match $PreferSheet }) | Select-Object -First 1
        }
        $p = @{ Path = $full; NoHeader = $true; DataOnly = $true }
        if ($Sheet) { $p.WorksheetName = $Sheet }
        $raw = @(Import-Excel @p)
        if ($raw.Count -eq 0) { throw "No data in $full" }
        $propNames = @($raw[0].PSObject.Properties.Name)
        $headers = @(foreach ($pn in $propNames) { [string]$raw[0].$pn })
        for ($r = 1; $r -lt $raw.Count; $r++) { $rows.Add(@(foreach ($pn in $propNames) { $raw[$r].$pn })) }
    }
    else {
        Write-Host "  ImportExcel module not found - using Excel COM" -ForegroundColor DarkYellow
        $xl = New-Object -ComObject Excel.Application
        $xl.Visible = $false; $xl.DisplayAlerts = $false
        try {
            $wb = $xl.Workbooks.Open($full, 0, $true)
            $ws = $null
            if ($Sheet) { $ws = $wb.Worksheets.Item($Sheet) }
            elseif ($PreferSheet) { foreach ($w in $wb.Worksheets) { if ($w.Name -match $PreferSheet) { $ws = $w; break } } }
            if (-not $ws) { $ws = $wb.Worksheets.Item(1) }
            $v = $ws.UsedRange.Value2
            if ($v -is [array]) {
                $nr = $v.GetLength(0); $nc = $v.GetLength(1)
                $headers = @(for ($c = 1; $c -le $nc; $c++) { [string]$v[1, $c] })
                for ($r = 2; $r -le $nr; $r++) {
                    $vals = New-Object object[] $nc
                    for ($c = 1; $c -le $nc; $c++) { $vals[$c - 1] = $v[$r, $c] }
                    $rows.Add($vals)
                }
            } elseif ($null -ne $v) { $headers = @([string]$v) }
            $wb.Close($false)
        }
        finally {
            $xl.Quit()
            [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($xl)
            [GC]::Collect(); [GC]::WaitForPendingFinalizers()
        }
    }
    return [pscustomobject]@{ Headers = $headers; Rows = $rows }
}

function Read-Sheet {
    param([string]$Path, [string]$Sheet, [string[]]$Required, [string[]]$Optional)
    $t = Read-Table -Path $Path -Sheet $Sheet
    $headers = $t.Headers
    $idx = @{}
    for ($i = 0; $i -lt $headers.Count; $i++) {
        $k = Normalize-Header $headers[$i]
        if ($k -and -not $idx.ContainsKey($k)) { $idx[$k] = $i }
    }
    $map = @{}
    foreach ($c in $Required) {
        $k = Normalize-Header $c
        if (-not $idx.ContainsKey($k)) { throw "Column '$c' not found in $Path. Found: $($headers -join ', ')" }
        $map[$c] = $idx[$k]
    }
    foreach ($c in $Optional) {
        $k = Normalize-Header $c
        if ($idx.ContainsKey($k)) { $map[$c] = $idx[$k] } else { Write-Host "  (optional column '$c' not found - left blank)" -ForegroundColor DarkGray }
    }
    $out = New-Object System.Collections.Generic.List[hashtable]
    foreach ($row in $t.Rows) {
        $h = @{}; $empty = $true
        foreach ($c in ($Required + $Optional)) {
            $s = if ($map.ContainsKey($c)) { To-Cell $row[$map[$c]] } else { '' }
            if ($s) { $empty = $false }
            $h[$c] = $s
        }
        if (-not $empty) { $out.Add($h) }
    }
    Write-Host ("  {0} rows, {1} columns (using {2})" -f $out.Count, $headers.Count, $map.Count)
    return , $out
}

    if (-not $agCanon.ContainsKey($x.Ag)) { $agCanon[$x.Ag] = $x.Ag }
    $x.Ag = $agCanon[$x.Ag]; $x.Unit = $x.Ag
}
$user   = @($all | Where-Object { $_.Cls -ne 'SNAPSHOT' })
$online = @($user | Where-Object { $_.Status -eq 'ONLINE' })
$offAll = @($user | Where-Object { $_.Status -ne 'ONLINE' -and ($_.Env -eq 'PROD' -or $_.Env -eq 'DR') })
$prod   = @($online | Where-Object { $_.Env -eq 'PROD' })
$dr     = @($online | Where-Object { $_.Env -eq 'DR' })
Write-Host ("Rows: {0} total | {1} snapshots skipped | {2} online PROD | {3} online DR | {4} not-online PROD/DR" -f `
    @($all).Count, (@($all).Count - $user.Count), $prod.Count, $dr.Count, $offAll.Count)

# ---------------------------------------------------------------- PROD -> DR pairing
$generic = New-Object 'System.Collections.Generic.HashSet[string]' ([StringComparer]::OrdinalIgnoreCase)
'dbautility','dba','dba_2','dba_sql_hardening','reportserver','reportservertempdb','testdb','test','testing','dba_test','dbaadmin','ssisdb','distribution' | ForEach-Object { [void]$generic.Add($_) }

$pairs = @{}
# AG: every DR server that hosts databases of the same AG (an AG can have several DR replicas)
foreach ($ag in @($prod | Where-Object { $_.Ag -ne 'NA' } | ForEach-Object { $_.Ag } | Sort-Object -Unique)) {
    $d = @($dr | Where-Object { $_.Ag -eq $ag } | ForEach-Object { $_.Short } | Sort-Object -Unique)
    if ($d.Count) { $pairs["AG|$ag"] = $d }
}
# Standalone: the DR server with the most non-generic database names in common, for the same applications
$drStand = @($dr | Where-Object { $_.Ag -eq 'NA' } | Group-Object Short)
foreach ($g in @($prod | Where-Object { $_.Ag -eq 'NA' } | Group-Object Short)) {
    $appSet = New-Object 'System.Collections.Generic.HashSet[string]' ([StringComparer]::OrdinalIgnoreCase)
    $names  = New-Object 'System.Collections.Generic.HashSet[string]' ([StringComparer]::OrdinalIgnoreCase)
    foreach ($x in $g.Group) {
        foreach ($a in $x.Apps) { [void]$appSet.Add($a) }
        if (-not $generic.Contains($x.Db)) { [void]$names.Add($x.Db) }
    }
    $best = $null; $bestScore = 0
    foreach ($h in $drStand) {
        $hn = New-Object 'System.Collections.Generic.HashSet[string]' ([StringComparer]::OrdinalIgnoreCase)
        foreach ($x in $h.Group) {
            $hit = $false; foreach ($a in $x.Apps) { if ($appSet.Contains($a)) { $hit = $true; break } }
            if ($hit) { [void]$hn.Add($x.Db) }
        }
        if ($hn.Count -eq 0) { continue }
        $hn.IntersectWith($names)
        if ($hn.Count -gt $bestScore) { $best = $h.Name; $bestScore = $hn.Count }
    }
    if ($best) { $pairs["SA|$($g.Name)"] = @($best) }
}
$drKey = @{}
foreach ($d in $dr) { $drKey["$($d.Short)|$($d.Db)".ToLowerInvariant()] = $d }

# ---------------------------------------------------------------- PROD rows (one per AG/instance + DB)
$copyRx = [regex]::new('(_\d{6,8}$|_\d{4}-\d\d-\d\dT|PITR|_bak$|_old$|_Arch_\d|_\d{2}[a-z]{3}\d{2}$|_mock$|_Price_archive|^XFER_)', 'IgnoreCase')
$groups = [ordered]@{}
foreach ($p in $prod) {
    $k = "$($p.Unit)|$($p.Db)"
    if (-not $groups.Contains($k)) { $groups[$k] = New-Object System.Collections.Generic.List[object] }
    $groups[$k].Add($p)
}
$prodOut = New-Object System.Collections.Generic.List[object]
foreach ($k in @($groups.Keys)) {
    $L = $groups[$k]; $f = $L[0]
    $pk = if ($f.Ag -ne 'NA') { "AG|$($f.Ag)" } else { "SA|$($f.Short)" }
    [string[]]$drs = @(@($pairs[$pk]) | Where-Object { $_ -and $drKey.ContainsKey("$_|$($f.Db)".ToLowerInvariant()) })
    $m = if ($drs.Count) { $drKey["$($drs[0])|$($f.Db)".ToLowerInvariant()] } else { $null }
    $mech = ''
    if ($m) {
        if ($f.Ag -ne 'NA') { $mech = 'Always On AG (DR replica)' }
        elseif ($m.Short -like '*-FAILOVER') { $mech = 'Azure SQL MI failover group' }
        else { $mech = 'Log shipping / backup-restore (inferred)' }
    }
    [string[]]$srvs = @($L | ForEach-Object { $_.Short } | Sort-Object -Unique)
    $apps = New-Object System.Collections.Generic.List[string]
    foreach ($x in $L) { foreach ($a in $x.Apps) { if (-not $apps.Contains($a)) { $apps.Add($a) } } }
    $prodOut.Add([ordered]@{
        app  = $apps[0]
        apps = $apps.ToArray()
        srv  = $srvs[0]
        srvs = $srvs
        db   = $f.Db
        mb   = [double](($L | Measure-Object -Property Mb -Maximum).Maximum)
        ag   = $f.Ag
        unit = $f.Unit
        dr   = $(if ($drs.Count) { $drs[0] } else { '' })
        drs  = $drs
        drmb = $(if ($m) { [double]$m.Mb } else { $null })
        mech = $mech
        copy = $copyRx.IsMatch($f.Db)
    })
}

# ---------------------------------------------------------------- offline list
$offOut = New-Object System.Collections.Generic.List[object]
foreach ($o in $offAll) {
    $offOut.Add([ordered]@{ app = $o.App; apps = [string[]]$o.Apps; srv = $o.Short; db = $o.Db; env = $o.Env; st = $o.Status; mb = [double]$o.Mb })
}

# ---------------------------------------------------------------- server specs + platform
$inv = @{}
foreach ($r in $invRows) {
    $s = Get-Short $r.ServerName
    if ($s -and -not $inv.ContainsKey($s)) { $inv[$s] = $r }
}
$serversOut = [ordered]@{}
foreach ($g in @($user | Group-Object Short)) {
    $s = $g.Name; $i = $inv[$s]
    $plat = @($g.Group | ForEach-Object { $_.Plat } | Where-Object { $_ }) | Select-Object -First 1
    if (-not $plat -and $i -and $i.Platform) { $plat = $i.Platform }
    $names = (@($g.Group | ForEach-Object { $_.Raw }) + $(if ($i) { @($i.ServerName) } else { @() })) -join ' '
    $vm = if ($i) { [string]$i.Size } else { '' }
    $pf = Get-Platform $plat $names $vm
    $envs = @($g.Group | ForEach-Object { $_.Env } | Sort-Object -Unique)
    $env = if ($envs -contains 'PROD') { 'PROD' } elseif ($envs -contains 'DR') { 'DR' } else { [string]($envs | Select-Object -First 1) }
    $ver = @($g.Group | ForEach-Object { $_.Ver } | Where-Object { $_ }) | Select-Object -First 1
    $ed = if ($i) { Get-Edition ("{0} {1}" -f $i.Edition, $i.SQLEdition) } else { '' }
    if (-not $ed) { $ed = @($g.Group | ForEach-Object { $_.Ed } | Where-Object { $_ }) | Select-Object -First 1 }
    $serversOut[$s] = [ordered]@{
        cpu  = $(if ($i) { To-Int $i.LogicalCPUCount } else { 0 })
        mem  = $(if ($i) { To-Int $i.PhysicalMemoryGB } else { 0 })
        vm   = $vm
        loc  = $(if ($i) { [string]$i.Location } else { '' })
        ver  = [string]$ver
        ed   = [string]$ed
        plat = [string]$plat
        pc   = $pf.Class
        pl   = $pf.Label
        env  = $env
    }
}
$noSpec = @($serversOut.Keys | Where-Object { -not $inv.ContainsKey($_) })
if ($noSpec.Count) { Write-Host ("  {0} servers not found in inventory (specs blank): {1}" -f $noSpec.Count, ($noSpec -join ', ')) -ForegroundColor DarkYellow }

# ---------------------------------------------------------------- optional restore history (embedded)
$histJson = 'null'
$histInfo = ''
if ($RestoreHistory) {
    $HistCols = @('RowType','ServerName','CollectedFrom','DatabaseName','SizeMB','DataMB','LogMB','State','RecoveryModel','LastRestoreDate',
        'RestoredFromServer','RestoredFromDatabase','RestoredBy','RestoreBackupMB','RestoreCompressedMB','RestoreStartTime','RestoreCompletedTime',
        'RestoreDurationSec','RestoreDuration','RestoreDurationSource','RestoreMBps','RestoreSeconds','RestoreDevice','RestoreFiles','SourceBackupSec',
        'SourceBackupMBps','RestoreCount','MeasuredRestores','LastFullBackupDate','LastFullBackupMB','LastFullBackupCompressedMB','LastFullBackupSec',
        'BackupMBps','BackupDevice','BackupFiles','BackupSamples','BackupMBpsMin','BackupMBpsMedian','BackupMBpsMax','HostCPU','HostMemoryGB',
        'InstantFileInit','VmType','SqlVersion','EngineEdition','SqlEdition')
    $t = Read-Table -Path $RestoreHistory -PreferSheet 'restore'
    $canon = @{}; foreach ($c in $HistCols) { $canon[(Normalize-Header $c)] = $c }
    $map = New-Object System.Collections.Generic.List[object]
    $used = @{}
    for ($i = 0; $i -lt $t.Headers.Count; $i++) {
        $k = Normalize-Header $t.Headers[$i]
        if ($canon.ContainsKey($k) -and -not $used.ContainsKey($k)) { $used[$k] = $true; $map.Add([pscustomobject]@{ I = $i; C = $canon[$k] }) }
    }
    if (-not ($used.ContainsKey('servername') -and $used.ContainsKey('databasename'))) {
        throw "Restore history file needs ServerName and DatabaseName columns. Found: $($t.Headers -join ', ')"
    }
    $sb = New-Object System.Text.StringBuilder
    [void]$sb.Append('{"name":').Append((Json-Str (Split-Path $RestoreHistory -Leaf))).Append(',"cols":[')
    [void]$sb.Append((@($map | ForEach-Object { Json-Str $_.C }) -join ',')).Append('],"rows":[')
    $n = 0
    foreach ($row in $t.Rows) {
        $cells = @(foreach ($m in $map) { To-Cell $row[$m.I] })
        if (-not ($cells -join '')) { continue }
        if ($n) { [void]$sb.Append(',') }
        [void]$sb.Append('[').Append((@($cells | ForEach-Object { Json-Str $_ }) -join ',')).Append(']')
        $n++
    }
    [void]$sb.Append(']}')
    $histJson = $sb.ToString()
    $histInfo = "$n rows from $(Split-Path $RestoreHistory -Leaf)"
    Write-Host "  restore history: $histInfo" -ForegroundColor Green
}

# ---------------------------------------------------------------- write HTML
$data = [ordered]@{ prod = $prodOut.ToArray(); off = $offOut.ToArray(); servers = $serversOut }
$json = (ConvertTo-Json -InputObject $data -Depth 6 -Compress).Replace('<', '<')
$enc = { param($s) [System.Net.WebUtility]::HtmlEncode($s) }
$src = (& $enc (Split-Path $DatabasesXlsx -Leaf)) + ' + ' + (& $enc (Split-Path $InventoryXlsx -Leaf))
$tokens = @{ DATA = $json; HIST = $histJson; GENDATE = (Get-Date -Format 'yyyy-MM-dd HH:mm'); SRCFILES = $src }
$html = [regex]::Replace($Template, '__(DATA|HIST|GENDATE|SRCFILES)__', { param($m) $tokens[$m.Groups[1].Value] })
$outFull = [System.IO.Path]::GetFullPath($OutHtml)
[System.IO.File]::WriteAllText($outFull, $html, (New-Object System.Text.UTF8Encoding($false)))

$withDr = @($prodOut | Where-Object { $_.dr }).Count
Write-Host ""
Write-Host ("PROD databases : {0}" -f $prodOut.Count) -ForegroundColor Green
Write-Host ("  with DR copy : {0}" -f $withDr) -ForegroundColor Green
Write-Host ("  no DR copy   : {0}" -f ($prodOut.Count - $withDr)) -ForegroundColor Green
Write-Host ("AGs            : {0}" -f @($prodOut | Where-Object { $_.ag -ne 'NA' } | ForEach-Object { $_.ag } | Select-Object -Unique).Count) -ForegroundColor Green
Write-Host ("Not online     : {0}" -f $offOut.Count) -ForegroundColor Green
if ($histInfo) { Write-Host ("History        : {0}" -f $histInfo) -ForegroundColor Green }
Write-Host ("Output         : {0}" -f $outFull) -ForegroundColor Green
if (-not $NoOpen) { Start-Process $outFull }
