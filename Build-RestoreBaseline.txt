.SYNOPSIS
  Builds the "Server Impact Restore Baseline" HTML (Restore Baseline, AG Architecture,
  Restore History tabs) from the latest Excel exports.
.DESCRIPTION
  Run with no parameters: two pop-ups ask for the Databases file, then the Inventory file.
  The HTML is saved next to the Databases file and opened in the browser.
  Reads only the columns it needs - extra columns in newer files are ignored.
  Column names are matched case-insensitively, ignoring spaces/underscores.
  Current OA Databases.xlsx  (required columns)
     ServerName, DatabaseName, DatabaseSizeMB, DatabaseStatus, Environment,
     ApplicationName, AGName, DatabaseClassification
     (optional) SQLVersion, Platform
  SQL OA Inventory.xlsx      (required) ServerName
     (optional) LogicalCPUCount, PhysicalMemoryGB, Size, Location
  Restore time labels:
     Actual    - measured restore duration from msdb + SQL error log (fixed)
     Estimated - database size / this database's last full backup MB/s (fixed)
     Simulated - shown for every database: overhead + size / server throughput.
                 Throughput defaults to the highest restore MB/s captured on the
                 server and can be changed per server with its Simulate button.
.EXAMPLE
  .\Build-RestoreBaseline.ps1
.EXAMPLE
  .\Build-RestoreBaseline.ps1 -DatabasesXlsx '.\Current OA Databases.xlsx' -InventoryXlsx '.\SQL OA Inventory.xlsx' -OutHtml .\Baseline.html
#>

/* =====================================================================
   Restore history baseline – ONE script for ALL servers (PROD + DR)
   One row per user database on this server. Read-only (msdb + error log).
   SQL Server 2008 R2 -> 2022 and Azure SQL Managed Instance.
   RestoreDurationSec = msdb restore start -> "Database was restored"
   entry for the same database in the SQL Server error log.
   ===================================================================== */
SET NOCOUNT ON;

/* 1. Number of error logs kept on this server */
IF OBJECT_ID('tempdb..#logs') IS NOT NULL DROP TABLE #logs;
CREATE TABLE #logs (ArchiveNo int, LogDate nvarchar(50), LogSizeBytes bigint);
BEGIN TRY INSERT #logs EXEC master.sys.xp_enumerrorlogs 1; END TRY BEGIN CATCH END CATCH;
DECLARE @maxLog int;
SELECT @maxLog = ISNULL(MAX(ArchiveNo), 6) FROM #logs;

/* 2. Restore completion entries from every error log */
IF OBJECT_ID('tempdb..#el') IS NOT NULL DROP TABLE #el;
CREATE TABLE #el (LogDate datetime, ProcessInfo nvarchar(64), [Text] nvarchar(max));
DECLARE @i int; SET @i = 0;
WHILE @i <= @maxLog
BEGIN
  BEGIN TRY INSERT #el EXEC master.dbo.xp_readerrorlog @i, 1, N'Database was restored'; END TRY BEGIN CATCH END CATCH;
  SET @i = @i + 1;
END;

;WITH r AS (            -- latest FULL restore per database
  SELECT rh.destination_database_name AS DatabaseName, rh.restore_date, rh.user_name,
         bs.server_name AS RestoredFromServer, bs.database_name AS RestoredFromDatabase,
         CAST(bs.backup_size/1048576.0 AS decimal(18,1)) AS BackupMB,
         ROW_NUMBER() OVER (PARTITION BY rh.destination_database_name ORDER BY rh.restore_date DESC) AS rn
  FROM msdb.dbo.restorehistory rh
  LEFT JOIN msdb.dbo.backupset bs ON bs.backup_set_id = rh.backup_set_id
  WHERE rh.restore_type = 'D'),
b AS (                  -- latest FULL backup per database
  SELECT database_name, backup_start_date, backup_finish_date, backup_size,
         ROW_NUMBER() OVER (PARTITION BY database_name ORDER BY backup_finish_date DESC) AS rn
  FROM msdb.dbo.backupset WHERE type = 'D' AND is_copy_only = 0)
SELECT ServerName            = CAST(SERVERPROPERTY('ServerName') AS sysname),
       DatabaseName          = d.name,
       SizeMB                = (SELECT SUM(CAST(mf.size AS bigint))*8/1024 FROM sys.master_files mf WHERE mf.database_id = d.database_id),
       [State]               = d.state_desc,
       LastRestoreDate       = ISNULL(CONVERT(varchar(19), r.restore_date, 120), 'No restore history'),
       RestoredFromServer    = r.RestoredFromServer,
       RestoredFromDatabase  = r.RestoredFromDatabase,
       RestoredBy            = r.user_name,
       RestoreBackupMB       = r.BackupMB,
       RestoreStartTime      = CONVERT(varchar(19), r.restore_date, 120),
       RestoreCompletedTime  = CONVERT(varchar(19), lc.CompletedAt, 120),
       RestoreDurationSec    = x.RestoreSec,
       RestoreDuration       = CAST(x.RestoreSec/3600 AS varchar(10)) + ':'
                             + RIGHT('0' + CAST((x.RestoreSec % 3600)/60 AS varchar(2)), 2) + ':'
                             + RIGHT('0' + CAST(x.RestoreSec % 60 AS varchar(2)), 2),
       RestoreDurationSource = CASE WHEN r.restore_date IS NULL   THEN 'No restore history'
                                    WHEN lc.CompletedAt IS NOT NULL THEN 'ErrorLog'
                                    ELSE 'Not in error log' END,
       RestoreMBps           = CAST(r.BackupMB / NULLIF(x.RestoreSec, 0) AS decimal(18,1)),
       LastFullBackupDate    = CONVERT(varchar(19), b.backup_finish_date, 120),
       LastFullBackupMB      = CAST(b.backup_size/1048576.0 AS decimal(18,1)),
       LastFullBackupSec     = DATEDIFF(SECOND, b.backup_start_date, b.backup_finish_date),
       BackupMBps            = CAST(b.backup_size/1048576.0
                                    / NULLIF(DATEDIFF(SECOND, b.backup_start_date, b.backup_finish_date), 0) AS decimal(18,1))
FROM sys.databases d
LEFT JOIN r ON r.DatabaseName = d.name AND r.rn = 1
LEFT JOIN b ON b.database_name = d.name AND b.rn = 1
OUTER APPLY (SELECT TOP 1 c.LogDate AS CompletedAt          -- completion entry for THIS database
             FROM #el c
             WHERE r.restore_date IS NOT NULL
               AND CHARINDEX(N'Database: ' + d.name + N',', c.[Text]) > 0
               AND c.LogDate >= r.restore_date
               AND c.LogDate <  DATEADD(DAY, 3, r.restore_date)
             ORDER BY c.LogDate) lc
OUTER APPLY (SELECT RestoreSec = DATEDIFF(SECOND, r.restore_date, lc.CompletedAt)) x
WHERE d.database_id > 4                 -- skip master, tempdb, model, msdb
  AND d.source_database_id IS NULL      -- skip database snapshots
ORDER BY d.name;

<# Restore history runner
   Runs Restore_History_All.sql on every PROD + DR server and writes ONE Excel file (.xlsx).
   Needs : SqlServer module   ->  Install-Module SqlServer -Scope CurrentUser
   Excel : ImportExcel module  ->  Install-Module ImportExcel -Scope CurrentUser
           (if ImportExcel is not installed, Microsoft Excel is used instead)
   Azure MI names may need the full FQDN (...database.windows.net). #>
$servers = @(
__SERVERLIST__
)
$here = if ($PSScriptRoot) { $PSScriptRoot } else { (Get-Location).Path }
$sql  = Get-Content (Join-Path $here 'Restore_History_All.sql') -Raw
$xlsx = Join-Path $here ("RestoreHistory_All_{0:yyyyMMdd_HHmm}.xlsx" -f (Get-Date))
$cols = 'ServerName','DatabaseName','SizeMB','State','LastRestoreDate','RestoredFromServer','RestoredFromDatabase','RestoredBy',
        'RestoreBackupMB','RestoreStartTime','RestoreCompletedTime','RestoreDurationSec','RestoreDuration','RestoreDurationSource',
        'RestoreMBps','LastFullBackupDate','LastFullBackupMB','LastFullBackupSec','BackupMBps'
$text = 'ServerName','DatabaseName','State','LastRestoreDate','RestoredFromServer','RestoredFromDatabase','RestoredBy',
        'RestoreStartTime','RestoreCompletedTime','RestoreDuration','RestoreDurationSource','LastFullBackupDate'

$rows = foreach ($s in $servers) {
  Write-Host "-> $s"
  try {
    $p = @{ ServerInstance = $s; Query = $sql; QueryTimeout = 600; ErrorAction = 'Stop' }
    if ((Get-Command Invoke-Sqlcmd).Parameters.ContainsKey('TrustServerCertificate')) { $p.TrustServerCertificate = $true }
    foreach ($r in @(Invoke-Sqlcmd @p)) {
      $o = [ordered]@{}
      foreach ($c in $cols) { $v = $r[$c]; $o[$c] = if ($v -is [DBNull]) { $null } else { $v } }
      [pscustomobject]$o
    }
  } catch { Write-Warning "$s : $($_.Exception.Message)" }
}
if (-not $rows) { Write-Warning 'No rows returned - nothing written.'; return }

if (Get-Module -ListAvailable -Name ImportExcel) {
  Import-Module ImportExcel
  $rows | Export-Excel -Path $xlsx -WorksheetName 'RestoreHistory' -TableName 'RestoreHistory' -AutoSize -FreezeTopRow -NoNumberConversion $text
} else {
  Write-Host 'ImportExcel not found - using Microsoft Excel to create the .xlsx'
  $tmp = [IO.Path]::ChangeExtension($xlsx, 'csv')
  $rows | Export-Csv $tmp -NoTypeInformation -Encoding UTF8
  $xl = New-Object -ComObject Excel.Application; $xl.DisplayAlerts = $false
  try { $wb = $xl.Workbooks.Open($tmp); $wb.SaveAs($xlsx, 51); $wb.Close($false) }
  finally { $xl.Quit(); [void][Runtime.InteropServices.Marshal]::ReleaseComObject($xl) }
  Remove-Item $tmp
}
Write-Host "Done -> $xlsx ($(@($rows).Count) rows)" -ForegroundColor Green

function estimate(ss,db,mb){const r=real(ss,db);if(!r)return null;const bm=num(r.LastFullBackupMB),bs=num(r.LastFullBackupSec);

function Normalize-Header([string]$h) { return (($h -replace '[^A-Za-z0-9]', '').ToLowerInvariant()) }

function Read-Sheet {
    param([string]$Path, [string]$Sheet, [string[]]$Required, [string[]]$Optional)
    $full = (Resolve-Path -LiteralPath $Path).Path
    Write-Host "Reading $full" -ForegroundColor Cyan
    $table = New-Object System.Collections.Generic.List[object]
    $headers = @()
    if (Get-Module -ListAvailable -Name ImportExcel) {
        Import-Module ImportExcel
        $p = @{ Path = $full; NoHeader = $true; DataOnly = $true }
        if ($Sheet) { $p.WorksheetName = $Sheet }
        $raw = @(Import-Excel @p)
        if ($raw.Count -eq 0) { throw "No data in $full" }
        $propNames = @($raw[0].PSObject.Properties.Name)
        $headers = @(foreach ($pn in $propNames) { [string]$raw[0].$pn })
        for ($r = 1; $r -lt $raw.Count; $r++) {
            $vals = @(foreach ($pn in $propNames) { $raw[$r].$pn })
            $table.Add($vals)
        }
    }
    else {
        Write-Host "  ImportExcel module not found - using Excel COM" -ForegroundColor DarkYellow
        $xl = New-Object -ComObject Excel.Application
        $xl.Visible = $false; $xl.DisplayAlerts = $false
        try {
            $wb = $xl.Workbooks.Open($full, 0, $true)
            $ws = if ($Sheet) { $wb.Worksheets.Item($Sheet) } else { $wb.Worksheets.Item(1) }
            $v = $ws.UsedRange.Value2
            $rows = $v.GetLength(0); $cols = $v.GetLength(1)
            $headers = @(for ($c = 1; $c -le $cols; $c++) { [string]$v[1, $c] })
            for ($r = 2; $r -le $rows; $r++) {
                $vals = New-Object object[] $cols
                for ($c = 1; $c -le $cols; $c++) { $vals[$c - 1] = $v[$r, $c] }
                $table.Add($vals)
            }
            $wb.Close($false)
        }
        finally {
            $xl.Quit()
            [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($xl)
            [GC]::Collect(); [GC]::WaitForPendingFinalizers()
        }
    }
    $idx = @{}
    for ($i = 0; $i -lt $headers.Count; $i++) {
        $k = Normalize-Header $headers[$i]
        if ($k -and -not $idx.ContainsKey($k)) { $idx[$k] = $i }
    }
    $map = @{}
    foreach ($c in $Required) {
        $k = Normalize-Header $c
        if (-not $idx.ContainsKey($k)) { throw "Column '$c' not found in $full. Found: $($headers -join ', ')" }
        $map[$c] = $idx[$k]
    }
    foreach ($c in $Optional) {
        $k = Normalize-Header $c
        if ($idx.ContainsKey($k)) { $map[$c] = $idx[$k] } else { Write-Host "  (optional column '$c' not found - left blank)" -ForegroundColor DarkGray }
    }
    $out = New-Object System.Collections.Generic.List[hashtable]
    foreach ($row in $table) {
        $h = @{}; $empty = $true
        foreach ($c in ($Required + $Optional)) {
            $val = if ($map.ContainsKey($c)) { $row[$map[$c]] } else { $null }
            $s = if ($null -eq $val) { '' } else { ([string]$val).Trim() }
            if ($s) { $empty = $false }
            $h[$c] = $s
        }
        if (-not $empty) { $out.Add($h) }
    }
    Write-Host ("  {0} rows, {1} columns (using {2})" -f $out.Count, $headers.Count, $map.Count)
    return , $out
}

function Get-Short([string]$name) {
    $n = "$name".Trim()
    if (-not $n) { return '' }
    $n = $n -replace ',\d+$', ''
    $inst = ''
    if ($n.Contains('\')) { $p = $n.Split('\', 2); $n = $p[0]; $inst = '\' + $p[1] }
    $n = $n -replace '\.database\.windows\.net$', ''
    return ($n.Split('.')[0] + $inst)
}

        plat = $f.Plat
    }
}
$noSpec = @($serversOut.Keys | Where-Object { -not $inv.ContainsKey($_) })
if ($noSpec.Count) { Write-Host ("  {0} servers not found in inventory (specs blank): {1}" -f $noSpec.Count, ($noSpec -join ', ')) -ForegroundColor DarkYellow }

# ---------------------------------------------------------------- write HTML
$data = [ordered]@{ prod = $prodOut.ToArray(); off = $offOut.ToArray(); servers = $serversOut }
$json = ConvertTo-Json -InputObject $data -Depth 6 -Compress
$json = $json.Replace('</', '<\/')
$enc = { param($s) [System.Net.WebUtility]::HtmlEncode($s) }
$src = (& $enc (Split-Path $DatabasesXlsx -Leaf)) + ' + ' + (& $enc (Split-Path $InventoryXlsx -Leaf))
$html = $Template.Replace('__DATA__', $json).Replace('__GENDATE__', (Get-Date -Format 'yyyy-MM-dd HH:mm')).Replace('__SRCFILES__', $src)
$outFull = [System.IO.Path]::GetFullPath($OutHtml)
[System.IO.File]::WriteAllText($outFull, $html, (New-Object System.Text.UTF8Encoding($false)))

$withDr = @($prodOut | Where-Object { $_.dr }).Count
Write-Host ""
Write-Host ("PROD databases : {0}" -f $prodOut.Count) -ForegroundColor Green
Write-Host ("  with DR      : {0}" -f $withDr) -ForegroundColor Green
Write-Host ("  no DR        : {0}" -f ($prodOut.Count - $withDr)) -ForegroundColor Green
Write-Host ("AGs            : {0}" -f @($prodOut | Where-Object { $_.ag -ne 'NA' } | ForEach-Object { $_.ag } | Select-Object -Unique).Count) -ForegroundColor Green
Write-Host ("Not online     : {0}" -f $offOut.Count) -ForegroundColor Green
Write-Host ("Output         : {0}" -f $outFull) -ForegroundColor Green
Start-Process $outFull
