#Requires -Version 5.1
<#
=========================================================================================
 SQL APPLICATION RESILIENCE AND MTTR FRAMEWORK
-----------------------------------------------------------------------------------------
 WHAT THIS PRODUCES
   One self-contained HTML file with four panels, read in this order:

     1. Framework & MTTR     The explanation. What the framework is, the three failure
                             scenarios, the replication mechanisms, how every restore
                             time was arrived at, and the assumptions you own.
     2. Application Overview The application at the top of a relationship map, then one
                             card per environment, then the SQL instances inside it, then
                             the databases on each instance. Click an instance name to
                             jump to its full recovery scope.
     3. SQL Instance MTTR    Pick a SQL instance and see every database on it and how
                             long the whole instance takes to bring back.
     4. Executive Deck       Work in progress. One screen that says what this is.

 COUNTS, NOT MARKERS
   Where a set of databases is summarised it is stated as a number and an environment
   spread - "6 databases, PROD 4 / UAT 1 / DEV 1" - never as a row of markers you would
   have to count yourself.

 TWO RULES THE WHOLE REPORT FOLLOWS
   1. Anything not known prints as NULL. Never as zero, never as an average, and never
      dressed up in a phrase. NULL is a fact: it says go and collect this.
   2. Recovery is counted one database at a time, one instance at a time. That is the
      longest it can take, and it is the only figure worth planning against.

 HOW A RESTORE TIME IS ARRIVED AT
   Every database resolves down a three-rung ladder and the rung is printed next to the
   number, everywhere the number appears:

     1. MEASURED   A restore that was actually timed on this database.
     2. DERIVED    This database's own measured backup throughput from msdb, multiplied
                   by a single named restore factor the operator owns.
     3. NULL       No timed restore and no backup history. Nothing is invented.

 INPUTS
   Required : an Excel workbook with sheets 'Prod_DBInventory' and 'NonProd_Inventory'.
              Any other sheet carrying an 'Application Name' column is read as extra
              per-application metadata and shown on the application panel.
   Optional : -CollectSqlEvidence connects read-only to each instance and reads msdb for
              backup and restore history, sys.databases for size and compatibility level,
              and the HA/DR configuration. Without it the report still builds and the
              unmeasured fields read NULL.

 SAFETY
   Every statement issued against SQL Server is a SELECT. Nothing is created, altered or
   dropped. Connections are tagged Application Name=SQL-MTTR-Framework so anyone asking
   "what is this session" can see.

 USAGE
   .\New-MttrFramework.ps1
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -CollectSqlEvidence
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -CollectSqlEvidence -TestConnectivity
   .\New-MttrFramework.ps1 -SourceWorkbook .\Estate.xlsx -Palette Slate    # unbranded copy
=========================================================================================
#>

function Select-ExcelWorkbook {
    Add-Type -AssemblyName System.Windows.Forms
    $dialog = New-Object System.Windows.Forms.OpenFileDialog
    $dialog.Title = 'Select the application / SQL inventory workbook'
    $dialog.Filter = 'Excel workbooks (*.xlsx;*.xlsm;*.xls)|*.xlsx;*.xlsm;*.xls|All files (*.*)|*.*'
    $dialog.FilterIndex = 1
    $dialog.CheckFileExists = $true
    $dialog.CheckPathExists = $true
    $dialog.Multiselect = $false
    $dialog.RestoreDirectory = $true
    if ($dialog.ShowDialog() -ne [System.Windows.Forms.DialogResult]::OK) {
        throw 'No Excel workbook was selected. The report was not generated.'
    }
    return $dialog.FileName
}

function ConvertTo-NormalizedName {
    param([AllowNull()][string]$Value)
    if ([string]::IsNullOrWhiteSpace($Value)) { return '' }
    return (($Value -replace '\s+', ' ').Trim())
}

function Get-HostAndPort {
    param([string]$Instance)
    $value = ConvertTo-NormalizedName $Instance
    $port = 1433
    if ($value -match ',\s*(\d+)\s*$') {
        $port = [int]$Matches[1]
        $value = $value -replace ',\s*\d+\s*$', ''
    }
    $hostName = ($value -split '\\')[0]
    [pscustomobject]@{ Host = $hostName; Port = $port }
}

function Test-TcpEndpoint {
    param([string]$HostName, [int]$Port, [int]$TimeoutMs)
    if ([string]::IsNullOrWhiteSpace($HostName)) { return 'Unknown' }
    $client = [System.Net.Sockets.TcpClient]::new()
    try {
        $task = $client.ConnectAsync($HostName, $Port)
        if (-not $task.Wait($TimeoutMs)) { return 'Disconnected' }
        if ($client.Connected) { return 'Connected' }
        return 'Disconnected'
    }
    catch { return 'Disconnected' }
    finally { $client.Dispose() }
}

function Read-ExcelWorksheet {
    param([Parameter(Mandatory)]$Workbook, [Parameter(Mandatory)][string]$WorksheetName)
    try { $sheet = $Workbook.Worksheets.Item($WorksheetName) }
    catch { throw "Worksheet '$WorksheetName' was not found in the workbook. Expected sheets named 'Prod_DBInventory' and 'NonProd_Inventory'." }

    $used = $sheet.UsedRange
    $values = $used.Value2
    if ($null -eq $values) { return @() }

    $rowCount = $used.Rows.Count
    $columnCount = $used.Columns.Count
    $headers = @{}
    for ($column = 1; $column -le $columnCount; $column++) {
        $name = ConvertTo-NormalizedName ([string]$values[1, $column])
        if ($name) { $headers[$name.ToLowerInvariant()] = $column }
    }

    $rows = [System.Collections.Generic.List[object]]::new()
    for ($rowNumber = 2; $rowNumber -le $rowCount; $rowNumber++) {
        $row = [ordered]@{}
        foreach ($header in $headers.Keys) {
            $row[$header] = ConvertTo-NormalizedName ([string]$values[$rowNumber, $headers[$header]])
        }
        if (($row.Values | Where-Object { $_ }).Count -gt 0) { $rows.Add([pscustomobject]$row) }
    }
    return $rows.ToArray()
}

function Read-ApplicationDetails {
    param([Parameter(Mandatory)]$Workbook, [string]$RequestedWorksheet = '')

    $excluded = @('prod_dbinventory', 'nonprod_inventory')
    $applicationHeaders = @('application name', 'application', 'app name', 'appname')
    $candidateSheets = [System.Collections.Generic.List[object]]::new()

    foreach ($sheet in @($Workbook.Worksheets)) {
        $sheetName = ConvertTo-NormalizedName ([string]$sheet.Name)
        if ($excluded -contains $sheetName.ToLowerInvariant()) { continue }
        if ($RequestedWorksheet -and $sheetName -ne $RequestedWorksheet) { continue }

        $used = $sheet.UsedRange
        if ($used.Rows.Count -lt 2 -or $used.Columns.Count -lt 2) { continue }
        $values = $used.Value2
        $headers = [System.Collections.Generic.List[object]]::new()
        $applicationColumn = 0

        for ($column = 1; $column -le $used.Columns.Count; $column++) {
            $displayName = ConvertTo-NormalizedName ([string]$values[1, $column])
            if (-not $displayName) { continue }
            $headers.Add([pscustomobject]@{ Name = $displayName; Column = $column })
            if ($applicationHeaders -contains $displayName.ToLowerInvariant()) { $applicationColumn = $column }
        }

        if ($applicationColumn -gt 0) {
            $candidateSheets.Add([pscustomobject]@{
                Sheet = $sheet; Name = $sheetName; Values = $values
                Headers = $headers.ToArray(); ApplicationColumn = $applicationColumn
                RowCount = $used.Rows.Count
            })
        }
    }

    if ($RequestedWorksheet -and $candidateSheets.Count -eq 0) {
        throw "Details worksheet '$RequestedWorksheet' was not found, or has no Application Name column."
    }

    $records = [System.Collections.Generic.List[object]]::new()
    foreach ($candidate in $candidateSheets) {
        for ($rowNumber = 2; $rowNumber -le $candidate.RowCount; $rowNumber++) {
            $application = ConvertTo-NormalizedName ([string]$candidate.Values[$rowNumber, $candidate.ApplicationColumn])
            if (-not $application) { continue }
            $fields = [System.Collections.Generic.List[object]]::new()
            foreach ($header in $candidate.Headers) {
                if ($header.Column -eq $candidate.ApplicationColumn) { continue }
                $value = ConvertTo-NormalizedName ([string]$candidate.Values[$rowNumber, $header.Column])
                if (-not $value) { $value = 'NULL' }
                $fields.Add([ordered]@{ label = $header.Name; value = $value; isNull = ($value -eq 'NULL') })
            }
            if ($fields.Count -gt 0) {
                $records.Add([ordered]@{ application = $application; sheet = $candidate.Name; fields = $fields.ToArray() })
            }
        }
    }
    return $records.ToArray()
}

function Get-FieldValue {
    param($Row, [string[]]$Candidates)
    foreach ($candidate in $Candidates) {
        $property = $Row.PSObject.Properties[$candidate.ToLowerInvariant()]
        if ($null -ne $property -and -not [string]::IsNullOrWhiteSpace([string]$property.Value)) {
            return ConvertTo-NormalizedName ([string]$property.Value)
        }
    }
    return ''
}

# =========================================================================================
# REGION  Read-only SQL Server evidence collection
#
#   Everything here degrades. One query failing on one instance costs you that one fact,
#   not the instance, and never the run. What could not be read is reported as
#   "not collected", which is a different statement from "nothing there".
# =========================================================================================

function Invoke-SqlRead {
    param(
        [Parameter(Mandatory)][string]$Instance,
        [string]$Database = 'master',
        [Parameter(Mandatory)][string]$Query,
        [int]$TimeoutSeconds = 20,
        [pscredential]$Credential
    )

    $connectionString = "Server=$Instance;Database=$Database;Application Name=SQL-MTTR-Framework;Connect Timeout=$TimeoutSeconds;"
    if ($Credential) {
        $connectionString += ('User ID={0};Password={1};' -f $Credential.UserName, $Credential.GetNetworkCredential().Password)
    }
    else {
        $connectionString += 'Integrated Security=SSPI;'
    }

    $connection = New-Object System.Data.SqlClient.SqlConnection $connectionString
    try {
        $connection.Open()
        $command = $connection.CreateCommand()
        $command.CommandText = $Query
        $command.CommandTimeout = $TimeoutSeconds
        $table = New-Object System.Data.DataTable
        $reader = $command.ExecuteReader()
        $table.Load($reader)
        $reader.Close()

        $rows = [System.Collections.Generic.List[object]]::new()
        foreach ($dataRow in $table.Rows) {
            $item = [ordered]@{}
            foreach ($column in $table.Columns) {
                $value = $dataRow[$column.ColumnName]
                $item[$column.ColumnName] = $(if ($value -is [System.DBNull]) { $null } else { $value })
            }
            $rows.Add([pscustomobject]$item)
        }
        return $rows.ToArray()
    }
    finally { $connection.Dispose() }
}

# "A network-related or instance-specific error occurred" tells the reader nothing and is the
# single most common thing a first-time runner will see. Name the actual cause and the fix.
function Get-ConnectionDiagnosis {
    param([string]$Message)

    $text = $Message
    $marker = 'argument(s): "'
    $position = $text.IndexOf($marker)
    if ($position -ge 0) { $text = $text.Substring($position + $marker.Length).TrimEnd('"') }
    $text = ($text -replace '\s+', ' ').Trim()

    if ($text -match 'error: 26|Error Locating Server') {
        return 'Named instance could not be resolved. SQL Browser on UDP 1434 is usually blocked - use SERVER,PORT instead of SERVER\INSTANCE.'
    }
    if ($text -match 'error: 40|server was not found') {
        return 'Server not found, or not accepting remote connections. Check the name first, then the firewall on TCP 1433.'
    }
    if ($text -match 'Login failed') {
        return 'Reached the instance but the login was refused. Grant the account VIEW SERVER STATE and read access to msdb.'
    }
    if ($text -match 'timeout expired|pre-login|period of time') {
        return 'No response before the connection timeout expired. Raise -SqlTimeoutSeconds, or the instance is down.'
    }
    if ($text -match 'permission|denied|principal') {
        return 'Reached the instance but the account cannot read the evidence. Needs VIEW SERVER STATE and msdb db_datareader.'
    }
    if ($text.Length -gt 180) { $text = $text.Substring(0, 180) + '...' }
    return $text
}

function Get-SqlInstanceEvidence {
    param(
        [Parameter(Mandatory)][string]$Instance,
        [int]$TimeoutSeconds,
        [pscredential]$Credential,
        [int]$WindowDays,
        [string[]]$CommandLogDatabase
    )

    $evidence = [ordered]@{
        instance = $Instance; status = 'Unreachable'; message = ''
        machine = ''; version = ''; productLevel = ''; edition = ''
        isClustered = $false; isHadrEnabled = $false
        modules = [ordered]@{}
        databases = @{}
    }

    # --- Instance identity. If this fails the instance is genuinely unreachable. -------------
    try {
        $facts = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT
    CAST(SERVERPROPERTY('MachineName')    AS nvarchar(200)) AS MachineName,
    CAST(SERVERPROPERTY('ProductVersion') AS nvarchar(50))  AS ProductVersion,
    CAST(SERVERPROPERTY('ProductLevel')   AS nvarchar(50))  AS ProductLevel,
    CAST(SERVERPROPERTY('Edition')        AS nvarchar(200)) AS Edition,
    CAST(ISNULL(SERVERPROPERTY('IsClustered'),0)   AS int)  AS IsClustered,
    CAST(ISNULL(SERVERPROPERTY('IsHadrEnabled'),0) AS int)  AS IsHadrEnabled;
"@
        if ($facts.Count -gt 0) {
            $evidence.status = 'Collected'
            $evidence.machine = [string]$facts[0].MachineName
            $evidence.version = [string]$facts[0].ProductVersion
            $evidence.productLevel = [string]$facts[0].ProductLevel
            $evidence.edition = [string]$facts[0].Edition
            $evidence.isClustered = ([int]$facts[0].IsClustered -eq 1)
            $evidence.isHadrEnabled = ([int]$facts[0].IsHadrEnabled -eq 1)
        }
    }
    catch {
        $evidence.message = Get-ConnectionDiagnosis $_.Exception.Message
        return $evidence
    }

    function Add-DbFact {
        param($Store, [string]$Name, [hashtable]$Values)
        if ([string]::IsNullOrWhiteSpace($Name)) { return }
        $key = $Name.ToUpperInvariant()
        if (-not $Store.ContainsKey($key)) { $Store[$key] = [ordered]@{ name = $Name } }
        foreach ($entry in $Values.GetEnumerator()) { $Store[$key][$entry.Key] = $entry.Value }
    }

    # --- Size, compatibility level, recovery model, state -------------------------------------
    try {
        $databases = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  d.name                                                       AS DatabaseName,
        d.compatibility_level                                        AS CompatibilityLevel,
        d.recovery_model_desc                                        AS RecoveryModel,
        d.state_desc                                                 AS State,
        d.is_published, d.is_subscribed, d.is_merge_published,
        CAST(SUM(CAST(mf.size AS bigint)) * 8.0 / 1048576.0 AS decimal(18,2)) AS SizeGB
FROM    sys.databases d
LEFT JOIN sys.master_files mf ON mf.database_id = d.database_id
GROUP BY d.name, d.compatibility_level, d.recovery_model_desc, d.state_desc,
         d.is_published, d.is_subscribed, d.is_merge_published;
"@
        foreach ($database in $databases) {
            Add-DbFact $evidence.databases ([string]$database.DatabaseName) @{
                sizeGb        = $(if ($null -ne $database.SizeGB) { [double]$database.SizeGB } else { $null })
                compatibility = $(if ($null -ne $database.CompatibilityLevel) { [int]$database.CompatibilityLevel } else { $null })
                recoveryModel = [string]$database.RecoveryModel
                state         = [string]$database.State
                isReplicated  = (([int]$database.is_published -eq 1) -or ([int]$database.is_subscribed -eq 1) -or ([int]$database.is_merge_published -eq 1))
            }
        }
        $evidence.modules['Databases'] = 'Collected'
    }
    catch { $evidence.modules['Databases'] = "Failed: $($_.Exception.Message)" }

    # --- Measured backup throughput. This is the honest basis for a restore estimate. ---------
    #     Throughput is measured on this database, on this instance, on this storage.
    try {
        $backups = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  bs.database_name                                                       AS DatabaseName,
        MAX(bs.backup_finish_date)                                             AS LastFullBackup,
        COUNT(*)                                                               AS Samples,
        AVG(CAST(DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) AS bigint)) AS AvgSeconds,
        AVG(CAST(bs.backup_size / 1048576.0 AS float))                         AS AvgMegabytes
FROM    msdb.dbo.backupset bs
WHERE   bs.type = 'D'
  AND   bs.backup_finish_date > DATEADD(DAY, -$WindowDays, GETDATE())
  AND   DATEDIFF(SECOND, bs.backup_start_date, bs.backup_finish_date) > 0
  AND   bs.backup_size > 0
GROUP BY bs.database_name;
"@
        foreach ($backup in $backups) {
            $seconds = [double]$backup.AvgSeconds
            $megabytes = [double]$backup.AvgMegabytes
            $throughput = $null
            if ($seconds -gt 0 -and $megabytes -gt 0) { $throughput = [math]::Round($megabytes / $seconds, 2) }
            Add-DbFact $evidence.databases ([string]$backup.DatabaseName) @{
                lastFullBackup      = $(if ($null -ne $backup.LastFullBackup) { ([datetime]$backup.LastFullBackup).ToString('yyyy-MM-dd') } else { $null })
                backupSamples       = [int]$backup.Samples
                backupThroughputMBs = $throughput
                backupMinutes       = [math]::Round($seconds / 60.0, 1)
            }
        }
        $evidence.modules['Backup history'] = 'Collected'
    }
    catch { $evidence.modules['Backup history'] = "Failed: $($_.Exception.Message)" }

    # --- Has this database ever actually been restored, and when ------------------------------
    try {
        $restores = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  rh.destination_database_name AS DatabaseName,
        MAX(rh.restore_date)         AS LastRestore,
        COUNT(*)                     AS RestoreCount
FROM    msdb.dbo.restorehistory rh
WHERE   rh.restore_date > DATEADD(DAY, -$WindowDays, GETDATE())
GROUP BY rh.destination_database_name;
"@
        foreach ($restore in $restores) {
            Add-DbFact $evidence.databases ([string]$restore.DatabaseName) @{
                lastRestore  = $(if ($null -ne $restore.LastRestore) { ([datetime]$restore.LastRestore).ToString('yyyy-MM-dd') } else { $null })
                restoreCount = [int]$restore.RestoreCount
            }
        }
        $evidence.modules['Restore history'] = 'Collected'
    }
    catch { $evidence.modules['Restore history'] = "Failed: $($_.Exception.Message)" }

    # --- A genuinely timed restore, if anything on this estate logs one ------------------------
    #     Ola Hallengren's maintenance solution writes CommandLog with StartTime/EndTime.
    #     Where it exists this is the only truly measured restore duration available.
    $commandLogFound = $false
    foreach ($logDatabase in $CommandLogDatabase) {
        if ($commandLogFound) { break }
        try {
            $timed = Invoke-SqlRead -Instance $Instance -Database $logDatabase -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
IF OBJECT_ID('dbo.CommandLog') IS NOT NULL
    SELECT  cl.DatabaseName                                                   AS DatabaseName,
            COUNT(*)                                                          AS Samples,
            AVG(CAST(DATEDIFF(SECOND, cl.StartTime, cl.EndTime) AS bigint))   AS AvgSeconds,
            MAX(cl.EndTime)                                                   AS LastTimedRestore
    FROM    dbo.CommandLog cl
    WHERE   cl.CommandType LIKE 'RESTORE%'
      AND   cl.EndTime IS NOT NULL
      AND   cl.StartTime > DATEADD(DAY, -$WindowDays, GETDATE())
      AND   DATEDIFF(SECOND, cl.StartTime, cl.EndTime) > 0
    GROUP BY cl.DatabaseName;
"@
            if ($timed.Count -gt 0) {
                $commandLogFound = $true
                foreach ($entry in $timed) {
                    Add-DbFact $evidence.databases ([string]$entry.DatabaseName) @{
                        timedRestoreMinutes = [math]::Round(([double]$entry.AvgSeconds) / 60.0, 1)
                        timedRestoreSamples = [int]$entry.Samples
                        timedRestoreSource  = "CommandLog in [$logDatabase]"
                    }
                }
                $evidence.modules['Timed restores'] = "Collected from [$logDatabase]"
            }
        }
        catch { }
    }
    if (-not $commandLogFound -and -not $evidence.modules.Contains('Timed restores')) {
        $evidence.modules['Timed restores'] = 'No timed-restore log found'
    }

    # --- Recovery mechanism. This decides whether a restore is even the recovery path. ---------
    $mechanisms = @{}
    if ($evidence.isClustered) { $mechanisms['*'] = 'Failover Cluster Instance' }

    if ($evidence.isHadrEnabled) {
        try {
            $availabilityGroups = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT  DB_NAME(drs.database_id)   AS DatabaseName,
        ag.name                    AS AgName,
        ar.availability_mode_desc  AS AvailabilityMode
FROM    sys.dm_hadr_database_replica_states drs
JOIN    sys.availability_replicas ar ON ar.replica_id = drs.replica_id
JOIN    sys.availability_groups   ag ON ag.group_id  = ar.group_id
WHERE   drs.is_local = 1;
"@
            foreach ($group in $availabilityGroups) {
                $mode = [string]$group.AvailabilityMode
                $label = "Always On AG '$($group.AgName)'"
                if ($mode -match 'SYNCHRONOUS') { $label += ' (synchronous)' } elseif ($mode) { $label += ' (asynchronous)' }
                $mechanisms[([string]$group.DatabaseName).ToUpperInvariant()] = $label
            }
            $evidence.modules['Availability groups'] = 'Collected'
        }
        catch { $evidence.modules['Availability groups'] = "Failed: $($_.Exception.Message)" }
    }

    try {
        $logShipped = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT primary_database AS DatabaseName FROM msdb.dbo.log_shipping_primary_databases;
"@
        foreach ($shipped in $logShipped) {
            $key = ([string]$shipped.DatabaseName).ToUpperInvariant()
            if (-not $mechanisms.ContainsKey($key)) { $mechanisms[$key] = 'Log shipping (primary)' }
        }
    }
    catch { }

    try {
        $mirrored = Invoke-SqlRead -Instance $Instance -TimeoutSeconds $TimeoutSeconds -Credential $Credential -Query @"
SELECT DB_NAME(database_id) AS DatabaseName, mirroring_state_desc AS MirrorState
FROM   sys.database_mirroring WHERE mirroring_guid IS NOT NULL;
"@
        foreach ($mirror in $mirrored) {
            $key = ([string]$mirror.DatabaseName).ToUpperInvariant()
            if (-not $mechanisms.ContainsKey($key)) { $mechanisms[$key] = "Database mirroring ($($mirror.MirrorState))" }
        }
    }
    catch { }

    foreach ($key in @($evidence.databases.Keys)) {
        $mechanism = ''
        if ($mechanisms.ContainsKey($key)) { $mechanism = $mechanisms[$key] }
        elseif ($mechanisms.ContainsKey('*')) { $mechanism = $mechanisms['*'] }
        elseif ($evidence.databases[$key].Contains('isReplicated') -and $evidence.databases[$key]['isReplicated']) { $mechanism = 'Transactional / merge replication' }
        else { $mechanism = 'Backup and restore only' }
        $evidence.databases[$key]['recoveryMechanism'] = $mechanism
    }

    return $evidence
}

    [object[]]$applicationDetails = @(Read-ApplicationDetails -Workbook $workbook -RequestedWorksheet $DetailsWorksheet)
}
finally {
    if ($null -ne $workbook) { $workbook.Close($false) | Out-Null }
    if ($null -ne $excel) { $excel.Quit() }
    if ($null -ne $workbook) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($workbook) }
    if ($null -ne $excel) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($excel) }
    [GC]::Collect(); [GC]::WaitForPendingFinalizers()
}

$inventory = [System.Collections.Generic.List[object]]::new()
function Add-InventoryRows {
    param(
        [Parameter(Mandatory)][object[]]$Rows,
        [Parameter(Mandatory)][string]$DefaultEnvironment
    )
    foreach ($row in $Rows) {
        $application = Get-FieldValue $row @('application name', 'application', 'app name', 'appname')
        $instance = Get-FieldValue $row @('instance name', 'servername')
        $server = Get-FieldValue $row @('servername')
        $database = Get-FieldValue $row @('databasename', 'database name')
        $environment = Get-FieldValue $row @('environment')
        if (-not $environment) { $environment = $DefaultEnvironment }
        if (-not $server) { $server = (Get-HostAndPort $instance).Host }

        # Hand-captured values from the workbook. These fill gaps the SQL collection leaves,
        # and are overridden by anything actually measured on the instance.
        $sheetSize = Get-FieldValue $row @('database size', 'database size (gb)', 'size gb', 'size (gb)', 'size')
        $sheetCompat = Get-FieldValue $row @('compatibility level', 'compatibility')
        $sheetMechanism = Get-FieldValue $row @('ha/dr method', 'hadr method', 'dr method', 'data replication method', 'replication method')
        $sheetRestore = Get-FieldValue $row @('last restore duration (minutes)', 'restore duration (minutes)', 'measured restore minutes')
        $sheetVersion = Get-FieldValue $row @('sql version', 'sql server version', 'version')

        $inventory.Add([pscustomobject]@{
            Application     = $(if ($application) { $application } else { 'Unattributed' })
            Environment     = $(if ($environment) { $environment.ToUpperInvariant() } else { 'UNKNOWN' })
            Server          = $(if ($server) { $server } else { '' })
            Instance        = $(if ($instance) { $instance } else { '' })
            Database        = $(if ($database) { $database } else { '' })
            SheetSize       = $sheetSize
            SheetCompat     = $sheetCompat
            SheetMechanism  = $sheetMechanism
            SheetRestore    = $sheetRestore
            SheetVersion    = $sheetVersion
        })
    }
}
Add-InventoryRows -Rows @($prodRows) -DefaultEnvironment 'PROD'
Add-InventoryRows -Rows @($nonProdRows) -DefaultEnvironment 'NONPROD'

    Write-Host ("Collecting read-only evidence from {0} SQL instance(s)..." -f $instanceNames.Count) -ForegroundColor Cyan
    $index = 0
    foreach ($instanceName in $instanceNames) {
        $index++
        Write-Progress -Activity 'Collecting SQL evidence' -Status $instanceName -PercentComplete (($index / [math]::Max(1, $instanceNames.Count)) * 100)
        $evidence = Get-SqlInstanceEvidence -Instance $instanceName -TimeoutSeconds $SqlTimeoutSeconds -Credential $SqlCredential -WindowDays $EvidenceWindowDays -CommandLogDatabase $CommandLogDatabase
        $evidenceByInstance[$instanceName] = $evidence
        $colour = $(if ($evidence.status -eq 'Collected') { 'Green' } else { 'Yellow' })
        Write-Host ("  [{0,3}/{1}] {2,-40} {3}" -f $index, $instanceNames.Count, $instanceName, $evidence.status) -ForegroundColor $colour
    }
    Write-Progress -Activity 'Collecting SQL evidence' -Completed
}

function Get-DatabaseEvidence {
    param([string]$Instance, [string]$Database)
    if (-not $evidenceByInstance.ContainsKey($Instance)) { return $null }
    $store = $evidenceByInstance[$Instance].databases
    $key = $Database.ToUpperInvariant()
    if ($store.ContainsKey($key)) { return $store[$key] }
    return $null
}

function Get-EvidenceValue {
    param($Store, [string]$Key)
    if ($null -eq $Store) { return $null }
    if ($Store.Contains($Key)) { return $Store[$Key] }
    return $null
}

    $row = $_
    $connectivity = $(if ($row.Instance) { $statusByInstance[$row.Instance] } else { 'Unknown' })
    $dbEvidence = Get-DatabaseEvidence -Instance $row.Instance -Database $row.Database

    # Size: measured wins, then hand-captured, then nothing at all.
    $sizeGb = Get-EvidenceValue $dbEvidence 'sizeGb'
    if ($null -eq $sizeGb) {
        $parsed = 0.0
        if ($row.SheetSize -and [double]::TryParse(($row.SheetSize -replace '[^0-9\.]', ''), [ref]$parsed) -and $parsed -gt 0) { $sizeGb = $parsed }
        else { $sizeGb = $null }
    }

    $compatibility = Get-EvidenceValue $dbEvidence 'compatibility'
    if ($null -eq $compatibility) {
        $parsedCompat = 0
        if ($row.SheetCompat -and [int]::TryParse(($row.SheetCompat -replace '[^0-9]', ''), [ref]$parsedCompat) -and $parsedCompat -gt 0) { $compatibility = $parsedCompat }
        else { $compatibility = $null }
    }

    $mechanism = Get-EvidenceValue $dbEvidence 'recoveryMechanism'
    if (-not $mechanism) {
        if ($row.SheetMechanism) { $mechanism = $row.SheetMechanism } else { $mechanism = $null }
    }

    $timedRestore = Get-EvidenceValue $dbEvidence 'timedRestoreMinutes'
    $timedSource = Get-EvidenceValue $dbEvidence 'timedRestoreSource'
    if ($null -eq $timedRestore -and $row.SheetRestore) {
        $parsedRestore = 0.0
        if ([double]::TryParse(($row.SheetRestore -replace '[^0-9\.]', ''), [ref]$parsedRestore) -and $parsedRestore -gt 0) {
            $timedRestore = $parsedRestore
            $timedSource = 'Recorded in the workbook'
        }
    }

    [ordered]@{
        application         = $row.Application
        environment         = $row.Environment
        server              = $row.Server
        instance            = $row.Instance
        database            = $row.Database
        connectivity        = $connectivity
        sizeGb              = $sizeGb
        compatibility       = $compatibility
        recoveryModel       = Get-EvidenceValue $dbEvidence 'recoveryModel'
        state               = Get-EvidenceValue $dbEvidence 'state'

        backupThroughputMBs = Get-EvidenceValue $dbEvidence 'backupThroughputMBs'
        backupMinutes       = Get-EvidenceValue $dbEvidence 'backupMinutes'
        lastFullBackup      = Get-EvidenceValue $dbEvidence 'lastFullBackup'
        lastRestore         = Get-EvidenceValue $dbEvidence 'lastRestore'
        timedRestoreMinutes = $timedRestore
        timedRestoreSamples = Get-EvidenceValue $dbEvidence 'timedRestoreSamples'
        timedRestoreSource  = $timedSource
    }
})

   listed it, so a repeated row never adds a second restore.
--------------------------------------------------------------------------- */
const dbKey = r => `${String(r.instance).toUpperCase()}|${String(r.database).toUpperCase()}`;

const databases = (() => {
  const map = new Map();
  raw.forEach(r => {
    const k = dbKey(r);
    if (!map.has(k)) {
      map.set(k, { key:k, instance:r.instance, server:r.server, database:r.database,
        environments:new Set(), applications:new Set(),
        sizeGb:r.sizeGb, compatibility:r.compatibility, recoveryModel:r.recoveryModel,
        state:r.state, mechanism:r.mechanism,
        backupThroughputMBs:r.backupThroughputMBs, backupMinutes:r.backupMinutes,
        lastFullBackup:r.lastFullBackup, lastRestore:r.lastRestore,
        timedRestoreMinutes:r.timedRestoreMinutes, timedRestoreSamples:r.timedRestoreSamples,
        timedRestoreSource:r.timedRestoreSource, connectivity:r.connectivity });
    }
    const d = map.get(k);
    d.environments.add(r.environment);
    d.applications.add(r.application);
  });
  map.forEach(d => {
    d.environment = [...d.environments].sort((a, b) => envRank(a) - envRank(b))[0];
    d.apps = [...d.applications].sort(byName);
    d.shared = d.apps.length > 1;
  });
  return map;
})();

const instances = (() => {
  const map = new Map();
  databases.forEach(d => {
    if (!map.has(d.instance)) {
      map.set(d.instance, { instance:d.instance, server:d.server, databases:[], apps:new Set(), environments:new Set() });
    }
    const i = map.get(d.instance);
    i.databases.push(d);
    d.apps.forEach(a => i.apps.add(a));
    d.environments.forEach(e => i.environments.add(e));
  });
  map.forEach(i => {
    i.environment = [...i.environments].sort((a, b) => envRank(a) - envRank(b))[0];
    i.appList = [...i.apps].sort(byName);
    i.databases.sort((a, b) => byName(a.database, b.database));
    i.meta = instanceMeta.find(m => m.instance === i.instance) || null;
    i.siblings = 0;
  });
  /* Two named instances on one host are still two instances, but they share a machine.
     The list says so, so nobody reads them as unrelated. */
  const byServer = new Map();
  map.forEach(i => { const s = String(i.server || '').toUpperCase(); byServer.set(s, (byServer.get(s) || 0) + 1); });
  map.forEach(i => { i.siblings = (byServer.get(String(i.server || '').toUpperCase()) || 1) - 1; });
  return map;
})();

const applications = (() => {
  const map = new Map();
  raw.forEach(r => {
    if (!map.has(r.application)) map.set(r.application, { name:r.application, rows:[], dbKeys:new Set() });
    const a = map.get(r.application);
    a.rows.push(r);
    a.dbKeys.add(dbKey(r));
  });
  map.forEach(a => { a.dbs = [...a.dbKeys].map(k => databases.get(k)); });
  return map;
})();

/* ---------------------------------------------------------------------------
   The evidence ladder: measured, derived, or NULL. Nothing is invented.
--------------------------------------------------------------------------- */
function restoreOf(db) {
  if (db.timedRestoreMinutes != null) {
    const n = db.timedRestoreSamples || 1;
    return { minutes: db.timedRestoreMinutes, basis: 'measured', label: 'Measured',
             detail: `Timed restore, ${n} run${n === 1 ? '' : 's'}${db.timedRestoreSource ? ' - ' + db.timedRestoreSource : ''}` };
  }
  const factor = num(cfg.factor);
  if (db.backupThroughputMBs != null && db.sizeGb != null && factor !== null) {
    return { minutes: (db.sizeGb * 1024 / db.backupThroughputMBs / 60) * factor, basis: 'derived', label: 'Derived',
             detail: `${db.sizeGb.toFixed(1)} GB at a measured ${db.backupThroughputMBs} MB/s backup throughput, x ${factor} restore factor` };
  }
  return { minutes: null, basis: 'none', label: 'NULL',
           detail: 'No timed restore and no backup history for this database. Run collection to measure it.' };
}

const basisBadge = r => `<span class="badge b-${r.basis}" title="${esc(r.detail)}">${esc(r.label)}</span>`;

/* One database at a time. That is the longest it can take, and it is the figure to plan against. */
function restoreWorkload(dbs) {
  const parts = dbs.map(restoreOf);
  const known = parts.filter(p => p.minutes !== null).map(p => p.minutes);
  return {
    total: known.length ? sum(known) : null,
    counted: known.length,
    unquantified: parts.length - known.length,
    mix: { measured: parts.filter(p => p.basis === 'measured').length,
           derived:  parts.filter(p => p.basis === 'derived').length,
           none:     parts.filter(p => p.basis === 'none').length }
  };
}

/* Scenario 3 - the whole instance is gone. */
function instanceMttr(inst) {
  const work = restoreWorkload(inst.databases);
  const build = buildHours(inst.environment);
  const validation = num(cfg.validation);
  const complete = (work.total !== null && build !== null)
    ? work.total + build * 60 + (validation || 0) * 60 : null;
  return { ...work, buildMinutes: build === null ? null : build * 60, complete };
}

/* Scenario 2 - the application is gone. Only its own databases are in scope, even where
   the instance is shared, but the rebuild of each instance it sits on still applies. */
function applicationMttr(app, environmentFilter) {
  const dbs = app.dbs.filter(d => environmentFilter === 'ALL' || d.environments.has(environmentFilter));
  const byInstance = new Map();
  dbs.forEach(d => {
    if (!byInstance.has(d.instance)) byInstance.set(d.instance, []);
    byInstance.get(d.instance).push(d);
  });

  const legs = [...byInstance.entries()].sort((a, b) => byName(a[0], b[0])).map(([instanceName, instanceDbs]) => {
    const inst = instances.get(instanceName);
    const work = restoreWorkload(instanceDbs);
    const build = buildHours(inst.environment);
    const validation = num(cfg.validation);
    const total = (work.total !== null && build !== null)
      ? work.total + build * 60 + (validation || 0) * 60 : null;
    return { instance: inst, dbs: instanceDbs.slice().sort((a, b) => byName(a.database, b.database)),
             work, buildMinutes: build === null ? null : build * 60, total };
  });

  const totals = legs.map(l => l.total).filter(t => t !== null);
  const complete = totals.length === legs.length && legs.length > 0 ? sum(totals) : null;

  const mix = { measured: 0, derived: 0, none: 0 };
  legs.forEach(l => Object.keys(mix).forEach(k => mix[k] += l.work.mix[k]));

  return { dbs, legs, complete, mix,
           collateral: uniq(legs.flatMap(l => l.instance.appList)).filter(a => a !== app.name).sort(byName) };
}

          <td class="mono">${nz(d.lastFullBackup)}${d.backupThroughputMBs != null ? `<div class="note">${d.backupThroughputMBs} MB/s measured</div>` : ''}</td>
          <td class="mono">${nz(d.lastRestore)}</td>
          <td class="num mono"><b>${fmtH(r.minutes)}</b></td>
          <td>${basisBadge(r)}</td></tr>`;
      }).join('')}</tbody></table></div>
    <p class="note" style="margin-top:9px">Databases are listed by name. Hover any basis badge to see exactly how that figure was arrived at.</p>`;
}

   Executive deck - in working. It says what this is, and no more.
--------------------------------------------------------------------------- */
function renderDeck() {
  const allDbs = [...databases.values()];
  const sized = allDbs.filter(d => d.sizeGb != null);
  const totalGb = sized.length ? sum(sized.map(d => d.sizeGb)) : null;
  const withFigure = allDbs.filter(d => restoreOf(d).minutes !== null).length;

  const instList = [...instances.values()].map(i => ({ inst: i, m: instanceMttr(i) }));
  const quantified = instList.filter(x => x.m.complete !== null);
  const worstInstance = quantified.slice().sort((a, b) => b.m.complete - a.m.complete)[0] || null;

  const appList = [...applications.values()].map(a => ({ app: a, m: applicationMttr(a, 'ALL') }));
  const worstApp = appList.filter(x => x.m.complete !== null).sort((a, b) => b.m.complete - a.m.complete)[0] || null;

  const dbTimes = allDbs.map(d => ({ db: d, r: restoreOf(d) })).filter(x => x.r.minutes !== null);
  const worstDb = dbTimes.slice().sort((a, b) => b.r.minutes - a.r.minutes)[0] || null;
  const medianDb = dbTimes.length
    ? dbTimes.slice().sort((a, b) => a.r.minutes - b.r.minutes)[Math.floor(dbTimes.length / 2)] : null;

  document.getElementById('deckLead').textContent =
    `${applications.size} applications run on ${instances.size} SQL instances holding ${databases.size} databases `
    + `(${spreadText(allDbs)}). This is what it currently takes to bring each of those back, one database at a time. `
    + `${withFigure} of ${databases.size} databases have a restore figure behind them; the rest are NULL until they are collected.`;

  document.getElementById('deckKpis').innerHTML = [
    ['Applications in scope', applications.size, `${[...instances.values()].filter(i => i.appList.length > 1).length} SQL instances carry more than one`, 'var(--accent)'],
    ['Databases to recover', databases.size, spreadHtml(allDbs), 'var(--info)'],
    ['Longest single instance loss', worstInstance ? fmtH(worstInstance.m.complete) : NUL,
      worstInstance ? esc(worstInstance.inst.instance) : 'Enter the rebuild hours on the Framework panel', 'var(--crit)'],
    ['Production instances', [...instances.values()].filter(i => isProd(i.environment)).length, `of ${instances.size} total`, 'var(--high)']
  ].map(kpiHtml).join('');

  const scenarios = [
    { h: '1 &middot; A database is lost',
      big: medianDb ? fmtH(medianDb.r.minutes) : NUL,
      who: medianDb
        ? `Typical database. The longest is <b>${esc(worstDb.db.database)}</b> ${envTag(worstDb.db.environment)} at <b>${hoursText(worstDb.r.minutes)}</b>.`
        : 'No database has a restore time yet. Run collection to measure them.' },
    { h: '2 &middot; An application is lost',
      big: worstApp ? fmtH(worstApp.m.complete) : NUL,
      who: worstApp
        ? `Worst case is <b>${esc(worstApp.app.name)}</b>, across ${worstApp.m.legs.length} instance${worstApp.m.legs.length === 1 ? '' : 's'}.`
        : 'Enter the instance rebuild hours to quantify this scenario.' },
    { h: '3 &middot; A SQL instance is lost',
      big: worstInstance ? fmtH(worstInstance.m.complete) : NUL,
      who: worstInstance
        ? `<b>${esc(worstInstance.inst.instance)}</b> ${envTag(worstInstance.inst.environment)} &mdash; ${worstInstance.inst.databases.length} databases.`
        : 'Enter the instance rebuild hours to quantify this scenario.' }
  ];
  document.getElementById('deckScenarios').innerHTML = scenarios.map(s =>
    `<div class="scenario"><div class="sh">${s.h}</div><div class="sb"><div class="big">${s.big}</div><div class="who">${s.who}</div></div></div>`).join('');
}
